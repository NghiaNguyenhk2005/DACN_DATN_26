\section*{Phụ lục B. Bộ công cụ đánh giá khả dụng}
\label{apdx:khaosat}
\addcontentsline{toc}{section}{Phụ lục B. Bộ công cụ đánh giá khả dụng}

Phụ lục này cung cấp công cụ để tiến hành đánh giá khả dụng nêu tại
mục~\ref{sec:danhgiaux}: thiết kế buổi thử, kịch bản tác vụ, phiếu ghi kết quả
và cách tính bốn chỉ số hành vi; kèm danh sách kiểm tra khả năng tiếp cận và
danh sách kiểm tra ứng dụng web cài đặt được.

\subsection*{B.1. Thiết kế buổi thử}

\noindent\textbf{Đối tượng tham gia.} Nông dân và thành viên ban điều hành hợp
tác xã tại vùng trồng thí điểm --- nhóm có mức độ thành thạo công nghệ thấp
nhất trong các nhóm người dùng (mục~\ref{sec:doituongnguoidung}). Đây là phép
thử khó nhất: nếu nhóm này dùng được thì các nhóm còn lại không phải trở ngại.

\noindent\textbf{Cỡ mẫu.} Tối thiểu 5 người, mục tiêu 8 người. Với kiểm thử khả
dụng, 5 người đã đủ phát hiện phần lớn lỗi nghiêm trọng về giao diện.

\noindent\textbf{Vì sao đo hành vi thay vì dùng bảng câu hỏi.} Bảng câu hỏi
chuẩn hóa cho ra một điểm tổng về cảm nhận, nhưng không chỉ ra người dùng vướng
ở đâu; với nhóm nông dân, việc tự chấm thang điểm năm mức cho các nhận định
trừu tượng cũng dễ bị trả lời cho xong. Bốn chỉ số hành vi dưới đây đo trực
tiếp điều nhóm cần biết để sửa giao diện: người dùng có tự làm xong không, mất
bao lâu, cần giúp mấy lần và thao tác sai ở đâu.

\noindent\textbf{Nguyên tắc tiến hành:}
\begin{itemize}
    \item Người quan sát \textbf{không hướng dẫn thao tác}. Chỉ đọc bối cảnh tác
    vụ rồi quan sát. Khi người tham gia hỏi, trả lời bằng câu trung lập kiểu
    ``anh/chị thử xem trên màn hình có gì giúp được không''.
    \item Ghi thời điểm bắt đầu và kết thúc từng tác vụ, mỗi lần người tham gia
    phải nhờ giúp, mỗi thao tác sai phải làm lại, và \emph{nguyên văn} câu than
    phiền nếu có --- câu nói tự nhiên của người dùng thường chỉ thẳng ra vấn đề
    giao diện.
    \item Một tác vụ được tính là \textbf{hoàn thành không cần hỗ trợ} khi người
    tham gia tự làm xong mà người quan sát không phải chỉ dẫn thao tác cụ thể
    lần nào.
\end{itemize}

\subsection*{B.4. Cách tính bốn chỉ số}

\begin{itemize}
    \item \textbf{Tỷ lệ hoàn thành không cần hỗ trợ} của một tác vụ = số người tự
    hoàn thành / tổng số người thử. Đây là chỉ số đối chiếu với ngưỡng 80\% tại
    Phụ lục L.
    \item \textbf{Thời gian hoàn thành} báo cáo bằng trung vị, chỉ tính trên những
    người đã hoàn thành; dùng trung vị vì vài người rất chậm sẽ kéo lệch trung
    bình.
    \item \textbf{Số lần nhờ giúp} và \textbf{số lỗi phải làm lại} báo cáo bằng
    trung bình trên mỗi người cho từng tác vụ, kèm danh sách chỗ vướng xếp theo
    số người gặp --- danh sách này là đầu vào trực tiếp cho vòng sửa giao diện.
\end{itemize}

\subsection*{B.5. Kiểm tra khả năng tiếp cận tự động}

Chạy công cụ axe-core trên các màn hình chính --- trang chủ, chi tiết sản phẩm,
trang truy xuất, giỏ hàng, trang đăng sản phẩm của nhà cung cấp --- đối chiếu
chuẩn WCAG mức AA. Báo cáo số lỗi theo mức độ nghiêm trọng và số lỗi đã sửa.
Cần ghi rõ giới hạn: kiểm tra tự động chỉ phát hiện được một phần lỗi tiếp
cận --- nhà phát triển công cụ công bố con số khoảng 57\% số lỗi gặp trong thực
tế, còn nếu tính theo số tiêu chí thành công của WCAG thì tỷ lệ tự động kiểm
được thấp hơn nhiều. Vì vậy kết quả này bổ sung chứ không thay thế buổi thử với
người dùng thật.

\subsection*{B.6. Danh sách kiểm tra ứng dụng web cài đặt được}

Thực hiện trên tối thiểu hai điện thoại thật, một Android tầm trung và một iOS
(NFR7.3):
\begin{enumerate}
    \item Cài được nền tảng lên màn hình chính và mở lại từ biểu tượng.
    \item Nhận được thông báo đẩy khi một đơn hàng đổi trạng thái.
    \item Tắt mạng, ghi một bản nhật ký canh tác; bật mạng lại, bản ghi được gửi
    đủ một lần, không trùng.
    \item Tắt mạng, cập nhật trạng thái một đơn giao hàng; bật mạng lại, trạng
    thái được đồng bộ đúng.
\end{enumerate}
