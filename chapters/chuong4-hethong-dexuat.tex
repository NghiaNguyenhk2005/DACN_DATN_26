\chapter{Hệ thống đề xuất}
\label{chap:hethong}

Chương này mô tả hệ thống Farmery từ góc nhìn nghiệp vụ và yêu cầu: ngữ cảnh nghiệp vụ cùng các quy tắc, chính sách chi phối hoạt động của sàn (mục~\ref{sec:ngucanh}); tổng quan hệ thống, các vấn đề nghiên cứu và thách thức cần giải quyết (mục~\ref{sec:motahethong}); và bộ yêu cầu chức năng, phi chức năng, dữ liệu (mục~\ref{sec:yeucau}). Phần mô hình hóa và thiết kế chi tiết được trình bày tại Chương~\ref{chap:phantich}.

\section{Ngữ cảnh nghiệp vụ}
\label{sec:ngucanh}

Farmery hoạt động trong lĩnh vực phân phối nông sản tươi --- loại hàng hóa có hạn sử dụng ngắn, chất lượng khó kiểm chứng trước khi mua và chịu điều chỉnh của các quy định về thương mại điện tử, an toàn thực phẩm và bảo vệ dữ liệu cá nhân. Mục này mô tả tổ chức kinh doanh của sàn, các quy trình nghiệp vụ kèm quy tắc nghiệp vụ, và các chính sách có yếu tố pháp lý, đạo đức.

\subsection{Mô hình kinh doanh}
\label{sec:businessmodelcanvas}

Farmery vận hành đồng thời hai luồng thương mại trên cùng một tập nguồn cung là nông dân và hợp tác xã đã được xác thực. Hai luồng khác nhau ở một điểm căn bản: \textbf{ai sở hữu hàng hóa trong lúc giao dịch diễn ra}. Chính điểm này kéo theo toàn bộ khác biệt về cách định giá, nguồn doanh thu và bên chịu rủi ro.

\begin{itemize}
    \item \textbf{Luồng bán sỉ --- sàn nhiều người bán (3P):} nhà cung cấp tự đăng sản phẩm, tự công bố bảng giá sỉ theo bậc số lượng và tự thương lượng với doanh nghiệp thu mua. Hàng hóa thuộc sở hữu nhà cung cấp cho tới khi bàn giao cho bên mua; Farmery đóng vai trò trung gian kết nối, bảo đảm minh bạch và thu phí hoa hồng trên mỗi giao dịch thành công.
    \item \textbf{Luồng bán lẻ --- Farmery thu mua và bán lại (1P):} Farmery chủ động đặt mua theo lô từ nhà cung cấp, tiếp nhận quyền sở hữu hàng hóa, tự phân loại, đóng gói, định giá và bán lại cho người tiêu dùng cá nhân dưới gian hàng của chính nền tảng. Doanh thu đến từ chênh lệch giữa giá thu mua và giá bán lẻ.
\end{itemize}

\noindent\textbf{Vì sao ghép được hai luồng trên cùng một nền tảng.} Hai luồng phục vụ hai nhu cầu mà một nhà cung cấp nông sản khó đáp ứng cùng lúc. Với đơn sỉ khối lượng lớn, nhà cung cấp giao thẳng cho doanh nghiệp thu mua là hiệu quả nhất, nền tảng chỉ cần đứng giữa để bảo đảm minh bạch. Nhưng với đơn lẻ vài ki-lô-gam, phần lớn nông dân và hợp tác xã không đủ năng lực phân loại, đóng gói và xử lý từng đơn nhỏ --- nếu để họ tự làm thì trải nghiệm người tiêu dùng không thể ổn định. Luồng 1P giải quyết đúng khoảng trống đó: Farmery gánh phần việc mà đầu chuỗi làm không tốt, đổi lại phải bỏ vốn và chịu rủi ro tồn kho.

Hai luồng hỗ trợ nhau theo một chiều: khi một lô đăng bán sỉ sắp hết thời hạn bảo quản mà chưa có người mua, nhân viên thu mua có thể lập đơn thu mua lô đó với giá chiết khấu, vừa giảm thiệt hại cho nhà cung cấp vừa bổ sung hàng cho gian hàng bán lẻ. Chiều ngược lại không mở: Farmery không bán sỉ, kể cả phần hàng bán lẻ tồn dư --- nền tảng không dùng vị thế vận hành sàn để cạnh tranh với chính nhà cung cấp của mình. Hàng tồn dư được xả bằng giảm giá ở kênh bán lẻ; việc chào phần tồn dư cho khách sỉ được để ở hướng phát triển (Chương~\ref{chap:ketluan}).

\noindent\textbf{Trục hiện thực.} Báo cáo thiết kế đủ hai luồng, nhưng phần hiện thực tập trung vào trục bán lẻ đầu--cuối: tìm kiếm $\rightarrow$ giỏ hàng $\rightarrow$ thanh toán $\rightarrow$ theo dõi đơn $\rightarrow$ đổi trả $\rightarrow$ đánh giá. Ba mức hiện thực được nêu tại mục~\ref{sec:phamvi}; các yêu cầu chỉ dừng ở mức thiết kế được ghi chú ngay tại mục~\ref{sec:yeucauchucnang}.

\noindent\textbf{Các dòng doanh thu.} Farmery có hai dòng thu chính, ứng với hai luồng:
\begin{itemize}
    \item \textbf{Hoa hồng trên đơn sỉ (3P):} doanh thu dịch vụ trung gian, tính trên mỗi đơn sỉ hoàn tất; Farmery xuất hóa đơn dịch vụ cho nhà cung cấp.
    \item \textbf{Chênh lệch giá mua--bán lẻ (1P):} dòng thu chính của phần hiện thực, chịu điều kiện hòa vốn nêu ngay dưới đây.
\end{itemize}

Các khoản còn lại được xác định rõ để không lẫn với nguồn lãi:
\begin{itemize}
    \item \textbf{Cước giao hàng lẻ chỉ bù chi phí:} gồm cước 3PL nội thành và phần bù chặng trung chuyển; miễn cước khi đơn vượt ngưỡng cấu hình.
    \item \textbf{Không thu phí đăng ký, phí duy trì tài khoản hay phí xử lý đơn} với nhà cung cấp, để giảm rào cản gia nhập cho nông dân và hợp tác xã (mục~\ref{sec:doituongnguoidung}).
    \item \textbf{Không bán dịch vụ ``xác thực chứng nhận nhanh'':} nếu trả tiền để được duyệt nhanh hơn, nhãn chứng nhận không còn là tín hiệu độc lập với người bán --- trái với lập luận niềm tin của vấn đề V4 (mục~\ref{sec:vande}).
    \item \textbf{Mức phí không ghi cố định trong báo cáo:} tỷ lệ hoa hồng, cước và các ngưỡng nằm trong \emph{Biểu phí công bố} do quản trị viên hệ thống cấu hình (FR10.6) và công khai cho người dùng (Phụ lục D).
\end{itemize}

Gói hiển thị nổi bật có gắn nhãn ``Tài trợ'' và dịch vụ kho mở cho nhà cung cấp (thuê kho, đóng gói, kiểm định) được để ở hướng phát triển (Chương~\ref{chap:ketluan}); nguyên tắc tách bạch vị trí trả phí với kết quả tự nhiên vẫn giữ trong Phụ lục D phòng khi bật lại.

\noindent\textbf{Điều kiện hòa vốn của luồng 1P.} Gọi $g$ là giá thu mua một ki-lô-gam, $h$ là tỷ lệ hao hụt từ lúc nghiệm thu tới lúc bán ($0 \le h < 1$), $c_{\text{đg}}$ là chi phí đóng gói một ki-lô-gam, $c_{\text{vh}}$ là chi phí vận hành một đơn và $q$ là số ki-lô-gam trung bình mỗi đơn. Giá bán lẻ $p$ chỉ có lãi khi:
\[
p - g \;\ge\; \underbrace{\frac{h}{1-h}\,g}_{\text{hao hụt}} \;+\; \underbrace{c_{\text{đg}}}_{\text{đóng gói}} \;+\; \underbrace{\frac{c_{\text{vh}}}{q}}_{\text{vận hành mỗi đơn}}
\]
Cước giao không có trong vế phải vì đã thu đủ từ khách. Với tỷ lệ tổn thất sau thu hoạch 20--40\% của ngành rau quả (vấn đề V3, mục~\ref{sec:vande}), riêng số hạng hao hụt đã bằng 25--67\% giá thu mua. Vì vậy thiết kế luồng 1P xoay quanh việc giữ $h$ thấp: chỉ thu mua lượng bán được trong thời hạn bảo quản, xuất hàng theo FEFO, và rút ngắn thời gian hàng nằm trên đường bằng chuyến trung chuyển cố định (mục~\ref{sec:quytrinhnghiepvu}).



\subsection{Quy trình nghiệp vụ và quy tắc nghiệp vụ}
\label{sec:quytrinhnghiepvu}

Mỗi quy trình dưới đây được trình bày theo các bước tuần tự, có ghi rõ tác nhân thực hiện, điều kiện chuyển bước (quy tắc nghiệp vụ) và các trường hợp ngoại lệ. Sơ đồ hoạt động của từng quy trình được trình bày tại mục~\ref{subsec:mohinhquytrinh}.

\subsubsection*{Đăng ký và xác thực nhà cung cấp, kiểm duyệt chứng nhận}
\begin{enumerate}
    \item Nhà cung cấp (nông dân hoặc hợp tác xã) đăng ký tài khoản, nhập thông tin định danh (căn cước công dân hoặc mã số thuế đối với hợp tác xã), thông tin trang trại, vùng canh tác, và khai báo loại người bán dùng cho nghĩa vụ thuế: tự sản xuất nông sản, hộ hoặc cá nhân kinh doanh, doanh nghiệp hoặc hợp tác xã (FR2.4).
    \item Hệ thống gửi mã xác thực (OTP) qua số điện thoại hoặc email; nếu nhập sai quá 5 lần, tài khoản bị khóa tạm thời trong 30 phút.
    \item Nhà cung cấp thực hiện xác minh danh tính điện tử (eKYC --- FR1.4): tải ảnh chụp giấy tờ định danh và ảnh chân dung để hệ thống đối chiếu; kết quả được ghi vào trạng thái eKYC của gian hàng.
    \item Nhà cung cấp tải lên giấy chứng nhận VietGAP, GlobalGAP, hữu cơ (organic), OCOP hoặc HACCP (nếu có), kèm mã số chứng nhận, cơ quan cấp và ngày hết hạn.
    \item Kiểm duyệt viên xác minh theo ba tầng: hệ thống kiểm tra định dạng mã số vùng trồng; kiểm duyệt viên đối chiếu mã này với cổng công bố công khai của Bộ Nông nghiệp và Môi trường và ghi kết quả; giấy chứng nhận VietGAP, GlobalGAP, hữu cơ được duyệt hồ sơ thủ công. Đối chiếu gián tiếp qua hệ thống truy xuất của Bộ là hướng phát triển (Chương~\ref{chap:ketluan}).
    \item Hệ thống ghi nhận kết quả: nếu eKYC đạt, tài khoản chuyển trạng thái ``Đã xác thực'' và được kích hoạt quyền đăng bán, đồng thời nhãn chứng nhận tương ứng (nếu chứng nhận cũng được duyệt) hiển thị công khai; nếu không đạt ở bước nào, hệ thống gửi thông báo lý do và hướng dẫn bổ sung cho bước đó.
\end{enumerate}

\noindent\textbf{Ngoại lệ:}
\begin{itemize}
    \item Chứng nhận còn hiệu lực dưới 30 ngày: hệ thống tự động nhắc nhà cung cấp gia hạn; nếu hết hạn mà chưa gia hạn, nhãn chứng nhận trên sản phẩm liên quan tự động bị ẩn cho đến khi cập nhật (sản phẩm không bị gỡ bỏ).
    \item Phát hiện chứng nhận giả mạo qua xác minh hoặc khiếu nại: tài khoản bị khóa vĩnh viễn, ghi vào danh sách vi phạm, không được phép đăng ký lại bằng cùng thông tin định danh.
\end{itemize}

\subsubsection*{Đăng và kiểm duyệt sản phẩm, sinh mã QR truy xuất}
\begin{enumerate}
    \item Nhà cung cấp đã xác thực tạo sản phẩm mới với đầy đủ thông tin: tên, mô tả, hình ảnh, đơn vị tính, giá bán lẻ, giá bán sỉ theo bậc (nếu có), sản lượng và ngày dự kiến thu hoạch.
    \item Nhà cung cấp ghi nhật ký canh tác theo từng giai đoạn (gieo trồng, chăm sóc, thu hoạch), mỗi lần ghi gồm ngày tháng, hoạt động và hình ảnh minh chứng nếu có; các bản ghi được gắn với một lô sản phẩm (batch) cụ thể.
    \item Hệ thống kiểm duyệt tự động sơ bộ: kiểm tra hình ảnh không trùng lặp, mô tả không chứa từ khóa cấm, giá không bất thường so với mặt bằng chung của danh mục.
    \item Kiểm duyệt viên kiểm tra thủ công lần cuối đối với sản phẩm mới hoặc nhà cung cấp chưa có lịch sử giao dịch; nhà cung cấp có uy tín lâu năm có thể áp dụng cơ chế hậu kiểm để rút ngắn thời gian chờ duyệt.
    \item Hệ thống sinh mã QR truy xuất duy nhất cho từng lô sản phẩm, liên kết tới trang truy xuất công khai gồm hồ sơ nhà cung cấp, chứng nhận, nhật ký canh tác và các mốc vận chuyển liên quan. Lô được mở bán ngay sau bước duyệt nội bộ; đồng bộ lên hệ thống truy xuất nguồn gốc của Bộ Nông nghiệp và Môi trường (mục~\ref{subsec:gs1blockchain}) là hướng phát triển.
\end{enumerate}

\noindent\textbf{Điều kiện:} sản phẩm chỉ được duyệt và hiển thị công khai khi có tối thiểu một bản ghi nhật ký canh tác gần nhất và hình ảnh sản phẩm thực tế.

\noindent\textbf{Quy tắc bảo đảm sản phẩm đăng bán đúng là nông sản.} Vì nền tảng chuyên biệt cho nông sản chứ không phải sàn tổng hợp, việc nhà cung cấp đăng nhầm hoặc cố tình đăng sai loại hàng sẽ làm hỏng chính giá trị mà nền tảng cam kết. Hệ thống kiểm soát điều này bằng bốn lớp:

\begin{enumerate}
    \item[\textbf{L1.}] \textbf{Danh mục đóng.} Sản phẩm bắt buộc gắn vào một danh mục lá trong cây danh mục nông sản do quản trị viên hệ thống định nghĩa sẵn theo phạm vi tại mục~\ref{subsec:dinhnghianongsan}; nhà cung cấp không được tự tạo danh mục mới. Mỗi danh mục lá quy định sẵn tập đơn vị tính hợp lệ và các trường bắt buộc kèm theo (ngày thu hoạch, hạn bảo quản, vùng trồng), nên hàng không phải nông sản sẽ không tìm được chỗ để khai báo hợp lệ.
    \item[\textbf{L2.}] \textbf{Danh sách loại trừ.} Nền tảng công bố công khai danh sách nhóm hàng không được đăng --- thủy sản sống, thịt tươi, động vật sống, thực phẩm chức năng, thuốc bảo vệ thực vật, vật tư nông nghiệp và mọi hàng hóa phi thực phẩm --- hiển thị ngay tại biểu mẫu đăng sản phẩm để giảm lỗi do hiểu nhầm.
    \item[\textbf{L3.}] \textbf{Kiểm duyệt trước khi hiển thị.} Hệ thống kiểm tra tự động sơ bộ (ảnh trùng lặp, từ khóa cấm, giá lệch bất thường so với mặt bằng danh mục, dấu hiệu sai danh mục), sau đó kiểm duyệt viên kiểm tra thủ công. Với nhóm chế biến sâu, hồ sơ tự công bố sản phẩm là điều kiện bắt buộc để được duyệt.
    \item[\textbf{L4.}] \textbf{Hậu kiểm bằng phản hồi cộng đồng.} Người mua có thể báo cáo sản phẩm sai danh mục hoặc không phải nông sản ngay tại trang sản phẩm; báo cáo được đưa vào hàng đợi xử lý của kiểm duyệt viên.
\end{enumerate}

\noindent\textbf{Chế tài khi vi phạm} được áp dụng theo ba mức tăng dần: lần đầu ẩn sản phẩm và yêu cầu nhà cung cấp phân loại lại; tái phạm thì cảnh cáo và trừ điểm uy tín (mục~\ref{subsec:danhgianghiepvu}); cố tình đăng hàng thuộc danh sách loại trừ hoặc tái phạm nhiều lần thì khóa quyền đăng bán.

\noindent\textbf{Ngoại lệ:}
\begin{itemize}
    \item Thay đổi thông tin trọng yếu sau khi đã duyệt (giá thay đổi vượt 20\% hoặc mô tả chất lượng thay đổi): sản phẩm chuyển về trạng thái chờ duyệt lại.
    \item Thay đổi nhỏ (còn hàng/hết hàng): không yêu cầu duyệt lại.
\end{itemize}

\subsubsection*{Quy trình thu mua vào kho của nền tảng}

Quy trình này đứng trước quy trình bán lẻ và là điểm khác biệt căn bản so với một sàn thuần trung gian.

\begin{enumerate}
    \item Nhà cung cấp công bố lô hàng sắp thu hoạch kèm sản lượng dự kiến, ngày thu hoạch và giá chào.
    \item Nhân viên thu mua đối chiếu các chào hàng cùng loại trong kỳ, tham khảo dự báo giá hiển thị ngay tại màn hình lập đơn (FR12.2, mục~\ref{sec:thietkeAI}) và tồn kho hiện có để quyết định mua lô nào, số lượng bao nhiêu.
    \item Nhân viên thu mua lập đơn thu mua ghi rõ số lượng, đơn giá và thời điểm giao; \textbf{nhà cung cấp xác nhận đơn trên hệ thống}. Đơn chỉ có hiệu lực sau bước xác nhận này, và từ lúc đó phần sản lượng trong đơn được giữ chỗ, không còn chào bán sỉ được nữa.
    \item Nhà cung cấp tự đưa hàng về kho sơ chế tại vùng trồng (chặng đầu). Farmery thuê xe đi gom tại nông hộ chỉ được thiết kế.
    \item Nhân viên kho nghiệm thu chất lượng thực tế và nhập kho phần đạt yêu cầu. Phần nhập kho trở thành một \emph{dòng tồn} thuộc sở hữu của Farmery, tách khỏi phần nhà cung cấp còn giữ, nhưng vẫn mang nguyên mã lô, nhật ký canh tác và mã QR truy xuất của lô gốc.
    \item Hệ thống thanh toán cho nhà cung cấp theo sản lượng đã nghiệm thu, theo chính sách nghiệm thu và thanh toán thu mua đã công bố (Phụ lục U) --- nhà cung cấp nhận tiền tại bước này, không phải chờ hàng bán được.
    \item Nhân viên kho đóng gói hàng thành đơn vị bán lẻ trong \textbf{một bước}, in và dán tem gồm mã QR của lô và nội dung nhãn hàng hóa bắt buộc (tên hàng, đơn vị chịu trách nhiệm, xuất xứ, định lượng, ngày đóng gói, hạn sử dụng, hướng dẫn bảo quản); phần hao hụt khi đóng gói được ghi ngay vào sổ nhập xuất. Sơ chế, phân loại nhiều bước chỉ được thiết kế.
    \item Nhân viên thu mua đặt giá bán lẻ dựa trên giá vốn, tỷ lệ hao hụt thực tế và mặt bằng giá thị trường, theo điều kiện hòa vốn tại mục~\ref{sec:businessmodelcanvas}, sau đó mở bán trên gian hàng của nền tảng.
\end{enumerate}

\noindent\textbf{Ngoại lệ:}
\begin{itemize}
    \item Nhà cung cấp không xác nhận đơn thu mua trong thời hạn quy định: đơn tự hủy, sản lượng không bị giữ chỗ.
    \item Chất lượng thực tế không đạt so với mô tả: nhân viên kho có quyền từ chối nhận hoặc nhận với giá điều chỉnh đã thống nhất lại; phần bị từ chối được trả về nhà cung cấp, trường hợp lệch nghiêm trọng được ghi vào lịch sử uy tín của nhà cung cấp.
    \item Lô đã nhập kho tồn quá lâu hoặc cận hạn: hệ thống cảnh báo để nhân viên thu mua chủ động giảm giá xả hàng ở kênh bán lẻ.
    \item Lô đang rao bán sỉ sắp hết hạn bảo quản mà chưa có người mua: nhân viên thu mua có thể lập đơn thu mua lô đó với giá chiết khấu, theo đúng các bước trên.
\end{itemize}

\subsubsection*{Quy trình bán lẻ}

Ở luồng này, bên bán là chính nền tảng: hàng đã thuộc sở hữu của Farmery từ quy trình thu mua ở trên.

\begin{enumerate}
    \item Khách lẻ tìm sản phẩm bằng câu tự nhiên hoặc từ khóa (gõ không dấu, gõ sai vẫn tìm được), lọc theo danh mục, vùng trồng, chứng nhận hoặc mức giá (FR4.5, FR12.3).
    \item Khách lẻ xem chi tiết sản phẩm kèm tóm tắt truy xuất và các sản phẩm tương tự; có thể mở trang truy xuất nguồn gốc --- vẫn dẫn tới nhật ký canh tác của nhà cung cấp gốc --- trước khi quyết định mua. Sản phẩm đang hết hàng hoặc chưa vào mùa cho phép đăng ký nhận thông báo khi có hàng.
    \item Khách lẻ thêm sản phẩm vào giỏ hàng, số lượng giới hạn theo lượng còn lại trên dòng tồn của Farmery.
    \item Ở bước thanh toán, khách chọn địa chỉ, khung giờ giao và phương thức thanh toán; hệ thống hiển thị đủ tiền hàng, thuế, cước giao trước khi khách xác nhận đặt hàng. Thanh toán khi nhận hàng (COD) chỉ mở cho đơn dưới ngưỡng cấu hình và tài khoản chưa từng từ chối nhận hàng; ngoài hai điều kiện đó, đơn phải trả trước qua cổng thanh toán.
    \item Hệ thống giữ chỗ trên dòng tồn của Farmery thuộc lô tương ứng; thao tác giữ chỗ và trừ tồn thực hiện trong một giao dịch có khóa để tránh bán vượt khi nhiều khách cùng đặt trên một dòng tồn.
    \item Đơn đặt trước \emph{giờ chốt đơn} hằng ngày được nhân viên kho đóng gói, dán tem QR và xếp lên \emph{chuyến trung chuyển} về TP.HCM trong ngày; đơn sau giờ chốt vào chuyến kế tiếp. Đơn vị vận chuyển nội thành (3PL) nhận hàng ở đầu TP.HCM và giao theo khung giờ khách đã chọn.
    \item Khách theo dõi đơn trên một dòng thời gian tám trạng thái: Đã đặt $\rightarrow$ Đã thanh toán $\rightarrow$ Đang đóng gói $\rightarrow$ Lên chuyến trung chuyển $\rightarrow$ 3PL đã nhận $\rightarrow$ Đang giao $\rightarrow$ Đã giao $\rightarrow$ Hoàn tất (hoặc Khiếu nại); mỗi lần chuyển trạng thái có thông báo đẩy hoặc email.
    \item Khách kiểm hàng khi nhận. Khi hết thời hạn khiếu nại của nhóm hàng mà không có khiếu nại, đơn chuyển Hoàn tất; hệ thống phát hành hóa đơn điện tử cho đơn (bản hiện thực phát hành bản giả lập kèm mã tra cứu) và mở quyền đánh giá. Từ trang đơn hàng, khách có thể mua lại đúng các sản phẩm của đơn cũ.
\end{enumerate}

\noindent\textbf{Ngoại lệ:}
\begin{itemize}
    \item Thanh toán trực tuyến thất bại: đơn hàng giữ trạng thái chờ thanh toán tối đa 15 phút trước khi tự hủy và nhả phần tồn đã giữ chỗ.
    \item Khách hủy đơn trước khi đóng gói: nhả tồn đã giữ, hoàn tiền nếu đã trả trước; sau khi đóng gói thì không hủy được mà xử lý theo chính sách đổi trả.
    \item Người nhận từ chối nhận hàng COD không có lý do chất lượng: đơn chuyển trạng thái hoàn hàng, tài khoản bị ghi nhận lịch sử từ chối (mất quyền COD); nhân viên kho đánh giá lại hàng trả về trước khi nhập lại dòng tồn hoặc ghi hủy.
    \item Nhiều đơn đặt cùng lúc vượt tồn kho còn lại: hệ thống ưu tiên đơn xác nhận trước; các đơn còn lại tự động hủy kèm thông báo hết hàng.
    \item Hàng giao không đạt chất lượng: xử lý theo quy trình đổi trả và khiếu nại (mục quản lý tồn kho, giao hàng và đổi trả dưới đây); Farmery chịu trách nhiệm trực tiếp với khách lẻ, sau đó mới đối chiếu với nhà cung cấp gốc hoặc 3PL tùy nguyên nhân.
\end{itemize}

\subsubsection*{Đánh giá sản phẩm và nhà cung cấp}
\label{subsec:danhgianghiepvu}

\begin{enumerate}
    \item Người mua đã xác nhận nhận hàng (bước cuối quy trình bán lẻ ở trên) được quyền gửi đánh giá gồm điểm số (1--5 sao), bình luận và hình ảnh, trong khung thời gian quy định sau khi nhận hàng --- chỉ áp dụng cho đơn hàng đã xác nhận (verified purchase), không cho phép đánh giá khi chưa mua.
    \item Hệ thống kiểm duyệt tự động sơ bộ (từ khóa cấm, liên kết đáng ngờ) kết hợp các luật ngưỡng giám sát bất thường (Chương~\ref{chap:coso}) để phát hiện đánh giá giả hoặc thao túng uy tín.
    \item Đánh giá không bị nghi ngờ được công khai ngay; đánh giá bị nghi ngờ chuyển cho kiểm duyệt viên xem xét thủ công trước khi công khai hoặc ẩn.
    \item Hệ thống cập nhật điểm uy tín tổng hợp của nhà cung cấp --- trung bình có trọng số theo thời gian và số lượng đơn, không tính các đánh giá bị ẩn.
    \item Nhà cung cấp có thể phản hồi công khai một lần dưới mỗi đánh giá.
\end{enumerate}

\noindent\textbf{Thang đo điểm uy tín (đo lường):} điểm uy tín tổng hợp được chuẩn hóa về thang 0--100 (không dùng trực tiếp thang 1--5 sao của từng đánh giá đơn lẻ), tính bằng trung bình có trọng số của điểm sao các đánh giá hợp lệ trong 6 tháng gần nhất, trong đó đánh giá gần đây và đánh giá gắn với đơn hàng giá trị lớn hơn được gán trọng số cao hơn đánh giá cũ/đơn giá trị nhỏ. Ngưỡng áp dụng cơ chế hậu kiểm nhanh và ưu tiên hiển thị được đặt ở mức từ 80/100 trở lên.

\noindent\textbf{Căn cứ:} thang 0--100 giúp phân biệt mức độ uy tín rõ ràng hơn thang 1--5 sao, vốn dễ bị dồn cụm ở mức 4--5 sao do tâm lý người mua ngại để lại đánh giá tiêu cực \parencite{springer_article}; trọng số theo thời gian đảm bảo điểm phản ánh chất lượng gần đây thay vì phụ thuộc mãi vào lịch sử cũ (một nhà cung cấp từng tốt nhưng gần đây sa sút chất lượng không nên tiếp tục hưởng cơ chế ưu tiên); ngưỡng 80/100 được đặt ở mức cao vì cơ chế hậu kiểm nhanh làm giảm bớt kiểm soát trước khi đăng bán, nên chỉ nên áp dụng cho nhà cung cấp thực sự đáng tin cậy, nhằm hạn chế rủi ro bị lợi dụng.

\noindent\textbf{Điều kiện:} nhà cung cấp đạt điểm uy tín từ ngưỡng quy định trở lên được áp dụng cơ chế hậu kiểm nhanh khi đăng sản phẩm mới (đã nêu tại mục~\ref{sec:quytrinhnghiepvu}, Đăng và kiểm duyệt sản phẩm) và được ưu tiên hiển thị trong kết quả tìm kiếm.

\noindent\textbf{Ngoại lệ:}
\begin{itemize}
    \item Đánh giá bị từ chối sau kiểm duyệt thủ công: ẩn khỏi trang sản phẩm, thông báo lý do cho người mua, không tính vào điểm uy tín.
    \item Nhà cung cấp bị phát hiện thao túng đánh giá (mua đánh giá ảo, trả đũa khách hàng qua kênh ngoài hệ thống): xử lý theo quy trình vi phạm tại mục~\ref{sec:quytrinhnghiepvu} (Quản trị hệ thống).
\end{itemize}

\subsubsection*{Quy trình bán sỉ (B2B)}
\begin{enumerate}
    \item Nhà cung cấp công bố bảng giá bán sỉ theo bậc số lượng, số lượng tối thiểu (MOQ) và lịch giao dự kiến cho từng sản phẩm/lô hàng.
    \item Doanh nghiệp thu mua xem bảng giá công khai, gửi yêu cầu báo giá (RFQ) tham chiếu bảng giá hoặc yêu cầu tùy chỉnh nếu nhu cầu khác chuẩn (sản phẩm, số lượng, tiêu chuẩn chất lượng, lịch giao).
    \item Hệ thống kiểm tra số lượng yêu cầu có đạt MOQ của nhà cung cấp hay không; nếu chưa đạt, hệ thống gợi ý gộp đơn với doanh nghiệp khác cùng khu vực hoặc chuyển hướng sang kênh bán lẻ.
    \item Nhà cung cấp xác nhận hoặc điều chỉnh báo giá theo bậc, thời gian có thể đáp ứng và điều kiện thanh toán.
    \item Hai bên thương lượng trong kênh trao đổi gắn với chính yêu cầu báo giá đó (FR6.2) cho đến khi thống nhất số lượng, đơn giá và ngày giao.
    \item Hệ thống sinh hợp đồng điện tử từ mẫu chuẩn, điền đầy đủ các điều khoản đã thống nhất, bao gồm điều khoản phạt vi phạm và điều khoản thanh toán; hai bên xem bản nháp và có thể yêu cầu sửa, mỗi lần sửa tạo một phiên bản mới.
    \item Hai bên \emph{xác nhận giao kết} hợp đồng trong hệ thống bằng mã xác nhận dùng một lần; hệ thống lưu tài khoản, thời điểm và mã băm của đúng phiên bản đã xác nhận.
    \item Nhà cung cấp tự giao hoặc thuê 3PL giao hàng \textbf{trong một đợt} theo ngày đã ký, và xuất hóa đơn bán hàng cho doanh nghiệp thu mua.
    \item Doanh nghiệp thu mua xác nhận nhận hàng và chất lượng, sau đó thanh toán qua cổng thanh toán. Farmery ghi nhận hoa hồng của đơn và xuất hóa đơn dịch vụ cho nhà cung cấp.
    \item Nhà cung cấp theo dõi doanh thu, hoa hồng đã trừ và lịch sử bán sỉ trên trang đối soát của mình.
\end{enumerate}

\noindent\textbf{Phần chỉ thiết kế:} giao nhiều đợt trên một hợp đồng (FR6.5); ký quỹ trung gian cho hợp đồng giá trị lớn (FR7.3) --- tiền bên mua được giữ tại cổng thanh toán và chỉ giải ngân khi xác nhận nhận hàng, nền tảng không trực tiếp giữ tiền; và khấu trừ, nộp thuế thay cho nhà cung cấp là hộ hoặc cá nhân kinh doanh (FR7.4) theo pháp luật thuế về thương mại điện tử (mục~\ref{sec:boicanhchinhsach}). Ở bản hiện thực, niềm tin giữa hai bên dựa vào mã băm của hợp đồng đã xác nhận và mã QR truy xuất của lô.

\noindent\textbf{Ngoại lệ:}
\begin{itemize}
    \item Nhà cung cấp không giao đủ số lượng hoặc đúng ngày: áp dụng điều khoản phạt trong hợp đồng, ghi nhận vào lịch sử uy tín của nhà cung cấp.
    \item Doanh nghiệp thu mua chậm thanh toán: hệ thống nhắc nhở tự động, có thể tạm khóa quyền gửi yêu cầu báo giá mới cho đến khi hoàn tất thanh toán.
    \item Một trong hai bên muốn hủy hợp đồng trước ngày giao: xử lý theo điều khoản hủy đã ký kết từ đầu.
    \item Doanh nghiệp thu mua từ chối một phần hàng: hàng quay về dòng tồn của nhà cung cấp; tranh chấp xử lý theo hợp đồng, nhân viên vận hành đứng giữa khi cần (FR10.3).
\end{itemize}

\subsubsection*{Quản lý tồn kho theo dòng tồn}
\noindent\textbf{Lô, dòng tồn và sổ nhập xuất.} Một lô hàng là đơn vị truy xuất nguồn gốc: nó có một mã lô, một nhật ký canh tác và một mã QR. Nhưng hàng của cùng một lô có thể nằm ở nhiều nơi và thuộc về nhiều chủ, nên tồn kho không được ghi trên lô mà ghi trên \emph{dòng tồn}. Mỗi dòng tồn xác định ba điều: thuộc lô nào, ai là chủ sở hữu, và hàng đang nằm ở đâu. Ví dụ, một hợp tác xã có lô 1.000 kg đang rao bán sỉ; khi Farmery thu mua 300 kg thì lô đó có hai dòng tồn --- 700 kg của hợp tác xã, để tại hợp tác xã, chỉ bán được cho khách sỉ; và 300 kg của Farmery, để tại kho sơ chế, chỉ bán được cho khách lẻ. Việc thu mua là một giao dịch chuyển 300 kg từ dòng thứ nhất sang dòng thứ hai.

Mọi thay đổi số lượng trên một dòng tồn đều phải đi qua một bản ghi trong \emph{sổ nhập xuất}, nêu rõ nguồn giao dịch, số lượng, người thực hiện và chứng từ liên quan. Không có thao tác nào được sửa trực tiếp con số tồn. Bảng~\ref{tab:nguon-giaodich} ở Phụ lục~Q liệt kê toàn bộ nguồn giao dịch được thiết kế.

\noindent\textbf{Thu hồi lô} là lý do trực tiếp để gắn tồn kho với truy xuất nguồn gốc. Khi phát hiện chứng nhận giả hoặc vấn đề an toàn thực phẩm ở một lô, nhân viên vận hành thực hiện thu hồi: hệ thống khóa toàn bộ dòng tồn của lô dù chúng thuộc chủ nào và nằm ở đâu, truy ngược sổ nhập xuất để lập danh sách các đơn đã bán từ lô, thông báo cho người mua, và trang truy xuất của lô hiện cảnh báo thu hồi.

\noindent\textbf{Nguyên tắc quản lý.} Mọi dòng tồn được quản lý theo nguyên tắc FEFO (mục~\ref{subsec:khobai}): hệ thống ưu tiên gợi ý bán và xuất trước các lô sắp hết hạn, cảnh báo chủ dòng tồn khi số lượng xuống dưới ngưỡng hoặc sắp hết hạn mà chưa bán hết. Farmery không vận hành kho trung chuyển quy mô lớn mà chỉ duy trì kho sơ chế nhỏ tại vùng trồng, chứa hàng đã thu mua cho luồng bán lẻ; hàng bán sỉ do nhà cung cấp tự bảo quản tại chỗ.

Hai tình huống sau được xử lý bằng quy tắc nghiệp vụ, chưa thiết kế chi tiết:
\begin{itemize}
    \item \textbf{Chênh sản lượng sau thu hoạch:} lô được công bố theo sản lượng dự kiến; nếu thực thu thấp hơn, các cam kết (hợp đồng sỉ, đơn thu mua) được đáp ứng theo thứ tự thời điểm xác nhận, phần thiếu xử lý theo điều khoản của từng hợp đồng.
    \item \textbf{Hàng bán sỉ bị khách từ chối, giao thiếu hoặc thừa:} hàng quay về đúng dòng tồn mà nó xuất ra và áp dụng điều khoản phạt của hợp đồng.
\end{itemize}

\subsubsection*{Giao hàng đơn lẻ ba chặng}
Kho sơ chế ở Lâm Đồng cách khách lẻ ở TP.HCM khoảng ba trăm ki-lô-mét (mục~\ref{sec:phamvi}), nên đơn lẻ đi qua ba chặng với ba bên chịu trách nhiệm khác nhau:
\begin{enumerate}
    \item \textbf{Chặng đầu --- nông hộ về kho sơ chế:} thuộc quy trình thu mua; nhà cung cấp tự đưa hàng tới kho.
    \item \textbf{Chặng giữa --- Lâm Đồng về TP.HCM:} xe lạnh chạy tuyến cố định theo chuyến, do Farmery thuê và điều phối. Mỗi \emph{chuyến trung chuyển} có danh sách đơn (manifest) được lập sau giờ chốt đơn hằng ngày; nhân viên kho cập nhật lúc xe rời kho và lúc bàn giao tại đầu TP.HCM.
    \item \textbf{Chặng cuối --- giao tới khách:} đơn vị vận chuyển nội thành (3PL) nhận hàng tại điểm bàn giao, giao theo khung giờ khách đã chọn và đẩy trạng thái về hệ thống qua webhook.
\end{enumerate}
Phí giao đơn lẻ gồm cước 3PL nội thành và phần bù chặng giữa chia theo đơn, miễn khi đơn vượt ngưỡng cấu hình. Đơn sỉ không đi qua ba chặng này: nhà cung cấp tự giao hoặc thuê 3PL theo hợp đồng, trong một đợt. Đội giao hàng riêng của Farmery chỉ có ở hướng phát triển (Chương~\ref{chap:ketluan}).

\subsubsection*{Đổi trả, hoàn tiền và khiếu nại đơn lẻ}
Toàn văn chính sách kiểm hàng và chính sách đổi trả--hoàn tiền nằm ở Phụ lục U; phần dưới đây là các quy tắc mà hệ thống phải thực thi.
\begin{itemize}
    \item \textbf{Đồng kiểm khi nhận:} khách được kiểm hàng trước khi nhận; từ chối tại cửa vì hàng hỏng thì được hoàn 100\% kể cả cước giao.
    \item \textbf{Thời hạn khiếu nại theo nhóm hàng}, tính từ lúc giao thành công: rau lá 24 giờ, củ quả 48 giờ, hàng khô 7 ngày; các mốc này cấu hình được (FR10.6).
    \item \textbf{Đổi ý:} không áp dụng với hàng tươi; hàng khô còn nguyên niêm được đổi ý trong thời hạn của nhóm.
    \item \textbf{Hoàn tiền không cần trả hàng} khi giá trị khiếu nại dưới ngưỡng cấu hình --- hàng tươi trả về gần như không bán lại được, chi phí thu hồi lớn hơn giá trị hàng; hệ thống ghi một bút toán hủy trong sổ nhập xuất để tồn kho vẫn khớp.
    \item \textbf{Lỗi của 3PL:} Farmery xử lý với khách trước, rồi đòi bồi hoàn từ 3PL sau; khách không phải tự làm việc với đơn vị vận chuyển.
    \item \textbf{Đơn sỉ} xử lý theo điều khoản hợp đồng giữa hai bên.
\end{itemize}

Quy trình xử lý một yêu cầu đổi trả hoặc khiếu nại:
\begin{enumerate}
    \item Khách gửi yêu cầu từ trang đơn hàng trong thời hạn của nhóm hàng, kèm ảnh hoặc video minh chứng.
    \item Hệ thống ghi nhận và gửi xác nhận tiếp nhận; mọi trao đổi giữa khách và nhân viên vận hành diễn ra trong luồng trao đổi gắn với chính yêu cầu đó (FR9.1), có đính kèm ảnh.
    \item Hệ thống gợi ý phân loại nguyên nhân: chất lượng từ khâu cung ứng, đóng gói, vận chuyển, hoặc bảo quản sai từ phía người nhận.
    \item Nhân viên vận hành ra quyết định: hoàn tiền toàn phần, hoàn một phần, giao bù, hoặc từ chối khi bằng chứng không đủ, kèm lý do.
    \item Hệ thống thực hiện hoàn tiền qua cổng thanh toán và ghi nhận nguyên nhân vào lịch sử của nhà cung cấp hoặc của 3PL để đối chiếu về sau.
\end{enumerate}

\noindent\textbf{Cam kết thời gian xử lý}, đồng thời là ngưỡng đo được khi đánh giá: tiếp nhận khiếu nại trong 24 giờ, ra quyết định trong 3 ngày làm việc, hoàn tiền trong 7 ngày làm việc kể từ khi có quyết định. Khiếu nại quá hạn chưa xử lý được tự động chuyển lên mức ưu tiên cao (FR10.3).

\noindent\textbf{Ngoại lệ:}
\begin{itemize}
    \item Khiếu nại lặp lại nhiều lần trên cùng một lô hoặc cùng một nhà cung cấp trong thời gian ngắn: tự động gắn cờ để nhân viên vận hành rà soát lô đó, có thể dẫn tới thu hồi lô.
    \item Khiếu nại có dấu hiệu bất thường từ phía người mua (tần suất cao, minh chứng trùng lặp): chuyển xem xét thủ công trước khi hoàn tiền, không áp dụng hoàn tiền không cần trả hàng.
\end{itemize}

\subsubsection*{Quản trị hệ thống}
\begin{enumerate}
    \item Kiểm duyệt viên thực hiện kiểm duyệt định kỳ hồ sơ nhà cung cấp mới, sản phẩm mới hoặc có thay đổi trọng yếu, và các chứng nhận sắp hết hạn.
    \item Nhân viên vận hành giám sát các cảnh báo tự động từ hệ thống, bao gồm tỷ lệ khiếu nại bất thường và các dấu hiệu giao dịch đáng ngờ do luật ngưỡng kích hoạt (Chương~\ref{chap:coso}); ngưỡng của các luật này do quản trị viên hệ thống cấu hình.
    \item Vi phạm được xử lý theo quy trình phân cấp từ nhắc nhở, cảnh cáo công khai, tạm khóa cho đến khóa vĩnh viễn tài khoản, tùy theo mức độ và số lần vi phạm, có lưu vết đầy đủ lý do và quyết định --- vi phạm về nội dung đăng bán do kiểm duyệt viên xử lý, vi phạm phát sinh từ đơn hàng do nhân viên vận hành xử lý.
    \item Nhân viên vận hành truy xuất báo cáo thống kê định kỳ để hỗ trợ ra quyết định vận hành và báo cáo cho các bên liên quan.
\end{enumerate}

Ngoại lệ: khi phát hiện vi phạm nghiêm trọng như giả mạo chứng nhận hoặc gian lận, tin đăng và lô liên quan bị ẩn ngay, tài khoản bị khóa mà không qua các bước nhắc nhở trung gian, đồng thời toàn bộ đơn hàng đang xử lý của tài khoản đó được rà soát để bảo vệ quyền lợi của khách hàng đã đặt trước. Quyết định khóa hoặc hạn chế tài khoản người bán được thông báo kèm lý do ít nhất 5 ngày trước khi có hiệu lực, trừ khi cơ quan nhà nước có yêu cầu (mục~
ef{sec:boicanhchinhsach}).

\subsection{Chính sách, pháp lý và đạo đức}
\label{subsec:chinhsach}

Các chính sách dưới đây chi phối cách sàn vận hành. Chúng được cụ thể hóa ở hai nơi: Điều khoản dịch vụ là quy chế chung (Phụ lục D), và bộ chính sách công khai cho người dùng --- kiểm hàng, đổi trả--hoàn tiền, bảo mật dữ liệu cùng các chính sách khác --- ở Phụ lục U. Bảng đối chiếu từng văn bản pháp luật với nghĩa vụ và tác động thiết kế nằm tại mục~\ref{sec:boicanhchinhsach}.
\begin{itemize}
    \item \textbf{Tuân thủ pháp luật:} Farmery vừa là nền tảng kinh doanh trực tiếp (bán lẻ) vừa là nền tảng trung gian (sàn sỉ) theo Luật Thương mại điện tử 2025, nên chịu cả hai nhóm nghĩa vụ: công khai các chính sách giao dịch, xác thực danh tính người bán, kiểm duyệt nội dung trước khi hiển thị, lưu dữ liệu giao dịch; cùng với nghĩa vụ về dữ liệu cá nhân, hóa đơn điện tử, thuế và nhãn hàng hóa (NFR8).
    \item \textbf{Giao kết hợp đồng điện tử:} theo Luật Giao dịch điện tử 2023~\cite{luatgddt2023}, chữ ký số có giá trị pháp lý tương đương chữ ký tay (Điều 23), còn các hình thức xác nhận khác bằng phương tiện điện tử để thể hiện sự chấp thuận thì thực hiện theo pháp luật liên quan (Điều 22, khoản 4). Vì vậy báo cáo gọi thao tác bằng mã xác nhận trong hệ thống là \emph{xác nhận giao kết}, không gọi là ký điện tử; ký số qua tổ chức cung cấp dịch vụ chứng thực được để ở hướng phát triển.
    \item \textbf{Không trực tiếp giữ tiền của người dùng:} thanh toán và hoàn tiền đi qua cổng thanh toán trung gian được cấp phép; Farmery chỉ ghi nhận trạng thái (NFR3.4). Ký quỹ trung gian cho đơn sỉ giá trị lớn theo cùng nguyên tắc, ở mức thiết kế (FR7.3).
    \item \textbf{Minh bạch trong xếp hạng:} ưu tiên hiển thị theo uy tín là quyền lợi miễn phí, và tiêu chí xếp hạng kết quả tìm kiếm được công khai; nếu sau này bật gói hiển thị trả phí, vị trí đó phải gắn nhãn ``Tài trợ'' để không làm sai lệch tín hiệu chất lượng.
    \item \textbf{Bảo vệ người tiêu dùng trong đơn lẻ:} trước khi đặt hàng, khách thấy đủ tiền hàng, thuế, cước giao và chính sách đổi trả; sau khi đặt, xem lại được toàn bộ nội dung đơn từ tài khoản. Chính sách đổi trả hàng tươi (không đổi ý, thời hạn theo nhóm hàng) phải được công khai trước khi khách đặt hàng.
    \item \textbf{Chống gian lận và thao túng:} chỉ người đã mua mới được đánh giá; đánh giá nghi vấn được kiểm duyệt thủ công; vi phạm xử lý phân cấp, vi phạm nghiêm trọng (giả mạo chứng nhận, gian lận) bị khóa ngay và có lưu vết quyết định.
    \item \textbf{Toàn vẹn dữ liệu truy xuất:} nhật ký canh tác và chứng nhận đã công khai không được sửa trực tiếp, mọi đính chính phải tạo bản ghi mới (NFR9.1).
    \item \textbf{Quyền của chủ thể dữ liệu:} người dùng đồng ý với chính sách bảo mật trước khi mở tài khoản; hệ thống lưu lại lần đồng ý (phiên bản chính sách, thời điểm), cho phép xem, sửa và yêu cầu xóa dữ liệu cá nhân, trừ phần dữ liệu giao dịch pháp luật buộc phải lưu (NFR8.2, NFR8.5).
    \item \textbf{Sử dụng dữ liệu cho AI:} dữ liệu do nhà cung cấp cung cấp thuộc quyền sở hữu của nhà cung cấp; Farmery chỉ dùng để vận hành nền tảng và huấn luyện mô-đun AI, ở dạng ẩn danh/tổng hợp khi không cần gắn với danh tính (Phụ lục D, Điều 13).
\end{itemize}

\section{Mô tả hệ thống}
\label{sec:motahethong}

\subsection{Tổng quan về hệ thống}
\label{sec:luongvanhanh}

\subsubsection*{Farmery là gì}

Farmery là nền tảng thương mại điện tử chuyên biệt cho nông sản, vận hành đồng thời hai luồng đã mô tả tại mục~\ref{sec:businessmodelcanvas}: một sàn bán sỉ nơi nông dân và hợp tác xã tự bán cho doanh nghiệp thu mua, và một gian hàng bán lẻ nơi chính nền tảng thu mua hàng rồi bán lại cho người tiêu dùng. Điểm xuyên suốt cả hai luồng là mọi sản phẩm đều gắn với một lô cụ thể có nhật ký canh tác và chứng nhận đã qua kiểm duyệt, tra cứu được bằng mã QR công khai.

Để tránh hiểu nhầm về phạm vi: Farmery không phải sàn thương mại điện tử tổng hợp --- danh mục hàng hóa giới hạn trong nhóm nông sản đã định nghĩa tại mục~\ref{subsec:dinhnghianongsan}, sản phẩm ngoài nhóm này bị từ chối ngay ở khâu đăng bán; cũng không phải dịch vụ đi chợ hộ, vì nền tảng bán hàng đã thu mua sẵn theo lô chứ không mua hộ theo yêu cầu từng khách; và không tự vận hành hệ thống kho lạnh quy mô lớn mà chỉ duy trì kho sơ chế nhỏ tại vùng trồng; đơn lẻ đi bằng xe lạnh thuê theo chuyến cố định về TP.HCM rồi giao qua đơn vị vận chuyển nội thành, Farmery không có đội giao hàng riêng (mục~\ref{sec:quytrinhnghiepvu}).

\subsubsection*{Các phân hệ chức năng}

Nền tảng được tổ chức thành năm khối chức năng, mỗi khối gom các phân hệ có cùng mục đích nghiệp vụ và ứng với các nhóm yêu cầu chức năng tại mục~\ref{sec:yeucauchucnang}:

\begin{itemize}
    \item \textbf{Nền tảng chung} --- tài khoản và phân quyền, đồng ý dữ liệu, thông báo, quản lý tệp (FR1, FR11).
    \item \textbf{Nguồn cung} --- hồ sơ nhà cung cấp, quản lý chứng nhận, sản phẩm và truy xuất nguồn gốc theo lô, tồn kho (FR2--FR4).
    \item \textbf{Giao dịch} --- bán sỉ qua báo giá và hợp đồng, bán lẻ từ gian hàng của nền tảng, thu mua và quản lý kho, thanh toán và hóa đơn (FR5--FR7, FR13).
    \item \textbf{Sau bán} --- giao hàng ba chặng, đổi trả và khiếu nại kèm trao đổi theo đơn, đánh giá và điểm uy tín (FR8, FR9).
    \item \textbf{Vận hành và hỗ trợ ra quyết định} --- công cụ quản trị và cấu hình, các mô-đun AI (FR10, FR12).
\end{itemize}

\subsubsection*{Kênh truy cập}

Người dùng bên ngoài và nhân sự nội bộ dùng hai ứng dụng web riêng, cùng gọi một giao diện lập trình (API) của hệ thống:
\begin{itemize}
    \item \textbf{Ứng dụng mua hàng} --- cho khách vãng lai, khách lẻ, khách sỉ và nhà cung cấp. Ứng dụng được dựng theo dạng \emph{ứng dụng web cài đặt được} (Progressive Web App, PWA): không bắt buộc cài, chỉ cần mở đường dẫn là dùng được, nhưng cài được lên màn hình chính để nhận thông báo đẩy; nhà cung cấp ghi nhật ký tại ruộng vẫn thao tác được khi sóng yếu, dữ liệu được gửi khi có kết nối (NFR7.3). Trang truy xuất nguồn gốc và trang sản phẩm được dựng sẵn phía máy chủ để mở nhanh khi quét mã QR và để máy tìm kiếm lập chỉ mục được.
    \item \textbf{Ứng dụng nội bộ} --- cho sáu vai trò nhân sự nội bộ (Bảng~\ref{tab:vaitro-noibo}). Đăng nhập bắt buộc xác thực hai lớp, phiên làm việc ngắn, có thể giới hạn theo địa chỉ mạng (NFR3.6); giao diện đáp ứng để nhân viên kho và nhân viên hỗ trợ vùng dùng được trên điện thoại.
\end{itemize}
Tách hai ứng dụng giữ cho ứng dụng mua hàng nhẹ và bảo đảm các màn hình có quyền cao không bao giờ được tải xuống máy của người dùng bên ngoài. Ứng dụng di động gốc nằm ngoài phạm vi đồ án (Chương~\ref{chap:ketluan}). Hỗ trợ chung cho người dùng đi qua Zalo OA và liên lạc của nhân viên hỗ trợ vùng với nhà cung cấp, không phải một kênh truy cập hệ thống (mục~\ref{sec:phamvi}).

\begin{itemize}
    \item \textbf{Nhà cung cấp} --- nông dân, hợp tác xã: đăng ký và xác thực, đăng sản phẩm, ghi nhật ký canh tác, phản hồi yêu cầu báo giá từ khách sỉ, hoặc bán cả lô cho Farmery qua nhân viên thu mua; giao hàng và nhận thanh toán.
    \item \textbf{Khách sỉ} --- doanh nghiệp thu mua: gửi yêu cầu báo giá, thương lượng và giao kết hợp đồng với nhà cung cấp qua kênh do hệ thống trung gian hóa, theo dõi đơn sỉ giao một đợt.
    \item \textbf{Khách lẻ} --- người tiêu dùng cá nhân: tìm kiếm, tra cứu truy xuất, đặt hàng từ gian hàng của nền tảng, theo dõi đơn, yêu cầu đổi trả và đánh giá. Khách lẻ giao dịch với Farmery chứ không với nhà cung cấp.
    \item \textbf{Nhân sự nội bộ Farmery} --- sáu vai trò tại Bảng~\ref{tab:vaitro-noibo}: khảo giá và tạo đơn thu mua theo lô, nghiệm thu, đóng gói và điều phối chuyến trung chuyển, kiểm duyệt nhà cung cấp và sản phẩm, xử lý khiếu nại và đổi trả, hỗ trợ nhà cung cấp tại vùng trồng, và cấu hình hệ thống.
    \item \textbf{Hệ thống nền tảng:} xác thực OTP, kiểm duyệt tự động sơ bộ, sinh mã QR, phục vụ kết quả từ các mô-đun AI, xử lý thanh toán và điều phối dữ liệu giữa các tác nhân. Bản hiện thực kết nối năm hệ thống bên ngoài: cổng thanh toán, đơn vị vận chuyển nội thành, dịch vụ hóa đơn điện tử (giả lập), dịch vụ email và dịch vụ đẩy của trình duyệt. Năm hệ thống chỉ được thiết kế: dịch vụ SMS, TraceViet, tổ chức chứng thực chữ ký số, và đăng nhập qua Google hoặc Facebook.
\end{itemize}

Một nguyên tắc chi phối toàn hệ thống: nền tảng không cho phép bên bán và bên mua giao dịch ngoài tầm kiểm soát của mình. Mọi luồng tiền, dữ liệu chứng nhận và đánh giá đều đi qua hệ thống, vừa để đảm bảo tính minh bạch (liên hệ mục~\ref{subsec:reputation}), vừa để nhân sự nội bộ có đủ dữ liệu giám sát và xử lý vi phạm. Ở luồng bán lẻ, nguyên tắc này thể hiện theo cách mạnh hơn: Farmery không đứng giữa mà là một bên trong giao dịch, trực tiếp chịu trách nhiệm chất lượng hàng bán ra trước người tiêu dùng.

\subsubsection*{Luồng tiếp cận và tham gia hệ thống}
\label{sec:luongtiepcan}

Mục này làm rõ cách mỗi nhóm người dùng ở mục~\ref{sec:doituongnguoidung} biết đến, liên hệ và được cấp quyền tham gia Farmery --- trước khi đi vào chi tiết từng quy trình nghiệp vụ ở mục~\ref{sec:quytrinhnghiepvu}.

\paragraph*{Nhà cung cấp (nông dân/HTX)}
\begin{itemize}
    \item \textbf{Kênh tiếp cận:} giới thiệu qua hội nông dân/HTX địa phương, chương trình OCOP (Chương~\ref{chap:coso}), hoặc tự đăng ký trực tiếp trên nền tảng web Farmery.
    \item \textbf{Liên hệ hỗ trợ:} nhân viên hỗ trợ vùng (liên lạc qua hotline hoặc Zalo, ngoài hệ thống) hướng dẫn và khi cần nhập hộ thông tin trang trại và ghi nhật ký canh tác cho nhóm ít rành công nghệ.
    \item \textbf{Thủ tục:} quy trình đăng ký, xác thực OTP và kiểm duyệt chứng nhận --- chi tiết đầy đủ tại mục~\ref{sec:quytrinhnghiepvu} (Đăng ký và xác thực nhà cung cấp).
    \item \textbf{Kết quả:} tài khoản chuyển trạng thái ``Đã xác thực'', được gán vai trò \texttt{seller}, có thể đăng bán lẻ và/hoặc công bố bảng giá bán sỉ.
\end{itemize}

\paragraph*{Người mua lẻ (người tiêu dùng cá nhân)}
\begin{itemize}
    \item \textbf{Kênh tiếp cận:} tự đăng ký trên nền tảng web qua kênh quảng bá số (mạng xã hội, SEO) hoặc giới thiệu từ cộng đồng quan tâm nông sản sạch.
    \item \textbf{Liên hệ hỗ trợ:} thắc mắc về một đơn cụ thể được trao đổi ngay trong trang đơn hoặc trang khiếu nại của đơn đó; hỏi đáp chung qua Zalo OA và hotline.
    \item \textbf{Thủ tục:} đăng ký số điện thoại hoặc email, xác thực OTP --- không yêu cầu xác minh giấy tờ, có thể mua ngay sau khi xác thực.
    \item \textbf{Kết quả:} tài khoản được gán vai trò \texttt{buyer\_retail}, truy cập ngay quy trình bán lẻ (mục~\ref{sec:quytrinhnghiepvu}).
\end{itemize}

\paragraph*{Người mua sỉ (doanh nghiệp thu mua)}
\begin{itemize}
    \item \textbf{Kênh tiếp cận:} đăng ký tài khoản doanh nghiệp qua form riêng trên nền tảng, hoặc liên hệ trực tiếp đội ngũ kinh doanh (account manager) của Farmery để được tư vấn.
    \item \textbf{Liên hệ hỗ trợ:} account manager phụ trách khu vực/ngành hàng, hỗ trợ thiết lập điều kiện giao nhận và thanh toán phù hợp quy mô doanh nghiệp.
    \item \textbf{Thủ tục:} khai báo mã số thuế/giấy phép kinh doanh, đội ngũ Farmery xác minh hồ sơ trước khi cấp quyền xem bảng giá sỉ và gửi RFQ --- thời gian xử lý dài hơn nhóm mua lẻ do cần thẩm định.
    \item \textbf{Kết quả:} tài khoản được gán vai trò \texttt{buyer\_wholesale}, truy cập quy trình bán sỉ (mục~\ref{sec:quytrinhnghiepvu}).
\end{itemize}

\paragraph*{Nhân sự nội bộ Farmery}
\begin{itemize}
    \item \textbf{Kênh tiếp cận:} không có kênh đăng ký công khai --- tài khoản do quản trị viên hệ thống (\texttt{platform\_owner}) cấp phát nội bộ.
    \item \textbf{Thủ tục:} gán một hoặc nhiều vai trò nội bộ trong Bảng~\ref{tab:vaitro-noibo} đúng theo phạm vi công việc được giao, theo nguyên tắc đặc quyền tối thiểu (NFR3.1); nhân sự đăng nhập ứng dụng nội bộ bằng xác thực hai lớp (NFR3.6).
\end{itemize}

Khi hoàn tất thủ tục tham gia ở trên, mọi tài khoản đều chịu sự ràng buộc của Điều khoản dịch vụ (Phụ lục D) --- quy định đầy đủ quyền, nghĩa vụ, phí và cơ chế xử lý tranh chấp giữa các bên.

\subsubsection{Chân dung khách hàng đại diện (Customer Personas)}
\label{subsec:customer-personas}

Để cụ thể hóa các nhóm người dùng đã nêu, bốn chân dung đại diện (persona) được xây dựng cho bốn nhóm trực tiếp tạo ra hoặc tiêu thụ giá trị trên nền tảng. Mẫu trình bày theo cách tiếp cận persona hướng mục tiêu (goal-directed persona) của Cooper và cộng sự~\cite{cooper2014aboutface}: mỗi persona gồm nhân vật, bối cảnh sử dụng, mục tiêu, điểm đau và kỳ vọng. Phần kỳ vọng được ghi bằng lời của người dùng, chưa gắn với yêu cầu cụ thể nào --- các yêu cầu chỉ được rút ra ở bước phân tích tại mục~\ref{sec:yeucau}.

\begin{itemize}
    \item \textbf{Persona 1: Nhà cung cấp (nông dân, giám đốc hợp tác xã)}
    \begin{itemize}
        \item \textit{Nhân vật:} Bác Ba Hưng, 52 tuổi, Giám đốc HTX Nông nghiệp Rau sạch Cần Đước (Long An).
        \item \textit{Đặc điểm số \& Thiết bị:} Sử dụng điện thoại Android tầm trung, mạng 4G không ổn định; thường dùng Zalo trao đổi thông tin, ngại thao tác máy tính và biểu mẫu phần mềm phức tạp.
        \item \textit{Mục tiêu:} Tìm đầu ra ổn định, giảm phụ thuộc thương lái trung gian ép giá vụ mùa; có hai lựa chọn đầu ra --- tự bán sỉ cho chuỗi bán lẻ và bếp ăn công nghiệp, hoặc bán cả lô cho Farmery khi cần tiền ngay; minh bạch quy trình VietGAP của hợp tác xã để nâng giá trị nông sản.

\subsection{Các vấn đề nghiên cứu}
\label{subsec:vandenghiencuu}

Bên cạnh bốn vấn đề thực tiễn nêu tại mục~\ref{sec:vande}, việc xây dựng Farmery đặt ra các vấn đề nghiên cứu sau:
\begin{enumerate}
    \item[\textbf{NC1.}] \textbf{Thiết kế cơ chế uy tín chống thao túng:} tổng hợp đánh giá chủ quan thành điểm uy tín phản ánh đúng chất lượng gần đây, hạn chế hiện tượng dồn cụm điểm cao và đánh giá giả, trên nền lý thuyết bất cân xứng thông tin (mục~\ref{subsec:reputation}).
    \item[\textbf{NC2.}] \textbf{Truy xuất nguồn gốc bất biến không dùng blockchain:} bảo đảm tính bất biến của nhật ký canh tác đã công khai trên cơ sở dữ liệu tập trung theo nguyên lý định danh GS1, và đồng bộ được với hệ thống truy xuất nguồn gốc của Nhà nước mà không để việc bán hàng phụ thuộc vào hệ thống đó (mục~\ref{subsec:gs1blockchain}).
    \item[\textbf{NC3.}] \textbf{Dự báo giá nông sản với dữ liệu hạn chế:} xây dựng mô hình chuỗi thời gian có tính mùa vụ, dựa trên dữ liệu giá công khai khi lịch sử giao dịch của một sàn mới còn ít, để hỗ trợ quyết định thu mua.
    \item[\textbf{NC4.}] \textbf{Gợi ý sản phẩm khi thiếu dữ liệu tương tác (cold start):} kết hợp đặc trưng nông sản (mùa vụ, vùng trồng, chứng nhận) với hành vi người dùng.
    \item[\textbf{NC5.}] \textbf{Cân bằng giữa hai luồng trên cùng một nguồn cung:} phân bổ sản lượng của một lô giữa luồng bán sỉ và luồng bán lẻ, chống cam kết vượt sản lượng khi hợp đồng bán sỉ và đơn thu mua cùng giành một phần hàng, và xác định thời điểm thu mua lô sỉ cận hạn sang kênh bán lẻ.
\end{enumerate}
Ngưỡng chất lượng mục tiêu cho các vấn đề NC3--NC4 được đặt tại Bảng~\ref{tab:nguongAI}.

\subsection{Các thách thức và khó khăn cần được giải quyết}
\label{subsec:thachthuc}

Ngoài các khó khăn của từng nhóm người dùng đã nêu tại mục~\ref{sec:doituongnguoidung}, ba thách thức dưới đây mang tính hệ thống --- không thuộc riêng nhóm nào và chi phối toàn bộ thiết kế:

\begin{itemize}
    \item \textbf{Chống bán vượt khi hai luồng dùng chung nguồn cung:} hàng của một lô được tách thành các dòng tồn theo chủ sở hữu, nên tranh chấp xảy ra ở hai chỗ. Trên dòng tồn của nhà cung cấp, hợp đồng bán sỉ và đơn thu mua của Farmery cùng giành một lượng hàng --- đây là nơi hai luồng thực sự chạm nhau. Trên dòng tồn của Farmery, nhiều đơn bán lẻ đặt cùng lúc tranh nhau phần còn lại. Cả hai đều buộc hệ thống giữ chỗ và trừ tồn trong một giao dịch có khóa (NFR1.2, NFR4.1). Đây là ràng buộc kỹ thuật khó nhất và là hệ quả trực tiếp của việc chọn mô hình hai luồng.
    \item \textbf{Hệ thống phải chạy được khi chưa có dữ liệu:} các mô-đun AI cần hoạt động ngay từ khi nền tảng mới ra mắt --- lúc chưa có lịch sử tương tác, chưa có dữ liệu giá nội bộ --- rồi cải thiện dần theo dữ liệu tích lũy.
    \item \textbf{Phụ thuộc vào các bên không kiểm soát được:} chất lượng bảo quản lạnh nằm ở đối tác vận chuyển; tính xác thực của chứng nhận và mã số vùng trồng nằm ở cơ quan cấp, vốn chưa có giao diện tra cứu công khai cho bên ngoài, và việc kết nối với hệ thống truy xuất của Bộ còn phụ thuộc vào việc được cấp quyền; còn dòng tiền phải đi qua cổng thanh toán được cấp phép do ràng buộc pháp lý (mục~\ref{subsec:chinhsach}).
\end{itemize}

\section{Yêu cầu}
\label{sec:yeucau}

Toàn bộ yêu cầu trong mục này xuất phát từ nhu cầu thực tế của các nhóm người dùng đã phân tích tại mục~\ref{sec:doituongnguoidung}, chứ không từ danh sách tính năng của các nền tảng khác. Để làm rõ mạch suy ra đó, mỗi nhóm người dùng được phát biểu thành các user story dạng \emph{``Là ..., tôi muốn ..., để ...''}, rồi từ đó mới hình thành yêu cầu tương ứng.

\subsubsection*{User story}

Mỗi nhóm người dùng dưới đây được minh họa bằng hai user story tiêu biểu nhất --- những nhu cầu mà nếu không đáp ứng được thì nhóm đó không còn lý do dùng nền tảng. Danh sách đầy đủ cùng bảng truy vết sang mã yêu cầu nằm tại Phụ lục J.

\begin{itemize}
    \item \textbf{Nhà cung cấp:} ``Là nông dân, tôi muốn bán được cả lô ngay khi thu hoạch, để không phải lo hàng hỏng trong lúc chờ người mua.'' \quad ``Là hợp tác xã, tôi muốn chứng minh sản phẩm của mình đạt chuẩn, để bán được giá cao hơn hàng không có chứng nhận.''
    \item \textbf{Khách sỉ:} ``Là bếp trưởng nhà hàng, tôi muốn đặt hàng định kỳ với sản lượng ổn định, để không phải tìm nguồn lại mỗi tuần.'' \quad ``Là bộ phận thu mua của siêu thị, tôi muốn xác minh được chứng nhận của nhà cung cấp, để đáp ứng yêu cầu kiểm soát chất lượng nội bộ.''
    \item \textbf{Khách lẻ:} ``Là người tiêu dùng, tôi muốn biết rau mình mua trồng ở đâu và ai trồng, để yên tâm về nguồn gốc.'' \quad ``Là người tiêu dùng, tôi muốn có một nơi chịu trách nhiệm khi hàng không đạt, để không phải tự đòi từng người bán nhỏ lẻ.''
    \item \textbf{Nhân viên thu mua:} ``Là nhân viên thu mua, tôi muốn so sánh chào giá giữa các nhà cung cấp theo mùa vụ, để quyết định mua của ai với giá nào.'' \quad ``Là nhân viên thu mua, tôi muốn được cảnh báo sớm các lô sắp hết hạn, để kịp xả hàng trước khi lỗ.''
    \item \textbf{Kiểm duyệt viên và nhân viên vận hành:} ``Là kiểm duyệt viên, tôi muốn duyệt nhanh hồ sơ và sản phẩm, để nhà cung cấp mới không phải chờ lâu rồi bỏ đi.'' \quad ``Là nhân viên vận hành, tôi muốn thấy bức tranh tổng thể về giao dịch và tồn kho, để phát hiện vấn đề trước khi nó thành sự cố.''
    \item \textbf{Nhân viên kho và nhân viên hỗ trợ vùng:} ``Là nhân viên kho, tôi muốn ghi nhận nhập, xuất và hao hụt ngay tại chỗ bằng điện thoại, để số tồn trên hệ thống luôn khớp với hàng thật.'' \quad ``Là nhân viên hỗ trợ vùng, tôi muốn nhập hộ nhật ký canh tác cho bà con qua điện thoại, để họ không bỏ cuộc vì vướng thao tác.''
\end{itemize}

Phụ lục J trình bày hai bảng: bảng truy vết từ từng user story sang mã yêu cầu tương ứng, và bảng đối chiếu nhóm yêu cầu chức năng với nhóm người dùng sử dụng trực tiếp. Chi tiết từng yêu cầu được trình bày tại các mục con dưới đây.

\subsection{Yêu cầu chức năng}
\label{sec:yeucauchucnang}

Dựa trên phân tích đối tượng người dùng (mục~\ref{sec:doituongnguoidung}) và
quy trình nghiệp vụ (mục~\ref{sec:quytrinhnghiepvu}), hệ thống cần đáp ứng các
nhóm yêu cầu chức năng sau. Các mã FRx.x dùng để đối chiếu 1--1 với use case
tại mục~\ref{sec:sddchitiet} và phần hiện thực ở Chương~\ref{chap:hienthuc}.

\noindent\textbf{Mức độ ưu tiên, tính khả thi và cách kiểm thử.} Toàn bộ 13 nhóm yêu cầu chức năng được đánh giá theo ba thuộc tính: mức độ ưu tiên theo MoSCoW, tính khả thi kỹ thuật, và phương pháp kiểm thử tương ứng. Phân bố cho thấy \textbf{9 nhóm ở mức Must} --- đây là phần lõi không thể cắt, gồm tài khoản và phân quyền, hồ sơ nhà cung cấp và tồn kho, chứng nhận, sản phẩm và truy xuất nguồn gốc, hai luồng giao dịch, thanh toán, vận hành nền tảng, và thu mua cùng quản lý kho. Ba nhóm ở mức Should (vận chuyển và khiếu nại, thông báo, gợi ý sản phẩm cùng tìm kiếm theo ngữ nghĩa) và hai nhóm ở mức Could (tương tác người dùng, dự báo giá). Về tính khả thi, gợi ý sản phẩm và dự báo giá là hai hạng mục duy nhất được đánh giá ở mức Challenging (tìm kiếm theo ngữ nghĩa ở mức Medium vì dùng mô hình nhúng có sẵn), phản ánh đúng rủi ro đã nêu tại mục~\ref{subsec:vandenghiencuu} về việc thiếu dữ liệu ở giai đoạn đầu. Ma trận đầy đủ kèm phương pháp kiểm thử cho từng nhóm được trình bày tại Phụ lục M.

\noindent\textbf{Quy ước ghi chú.} Yêu cầu hoặc phần yêu cầu có ghi \emph{(chỉ thiết kế)} được đặc tả và thiết kế đầy đủ nhưng không hiện thực trong đồ án; các mục còn lại thuộc phần hiện thực. Năm mã của phiên bản trước --- FR4.8 đồng bộ TraceViet, FR6.6 ký số, FR11.3 cảnh báo khan hiếm, FR13.5 thuê kho, FR13.7 chào sỉ tồn dư --- đã được rút khỏi bộ yêu cầu và chuyển sang hướng phát triển (Chương~\ref{chap:ketluan}); các mã còn lại giữ nguyên số để không lệch truy vết.

\subsubsection*{FR1: Quản lý tài khoản \& phân quyền}
\begin{itemize}
    \item \textbf{FR1.1:} Đăng ký/đăng nhập theo vai trò riêng biệt: nông
    dân/HTX, người tiêu dùng, doanh nghiệp thu mua (B2B), nhân sự nội bộ;
    đăng nhập bằng email hoặc số điện thoại. Đăng nhập bằng tài khoản mạng
    xã hội (OAuth 2.0) \emph{(chỉ thiết kế)}.
    \item \textbf{FR1.2:} Xác thực tài khoản qua OTP (đăng ký, khôi phục mật
    khẩu, các thao tác nhạy cảm).
    \item \textbf{FR1.3:} Đăng ký tài khoản doanh nghiệp thu mua (B2B) kèm
    xác thực Giấy phép kinh doanh (GPKD) và Mã số thuế (MST).
    \item \textbf{FR1.4:} Quy trình xác minh danh tính người bán (eKYC) đối
    với nông dân/HTX trước khi kích hoạt quyền đăng bán.
    \item \textbf{FR1.5:} Quản lý sổ địa chỉ giao/nhận hàng (nhiều địa chỉ,
    gán nhãn, chọn địa chỉ mặc định).
    \item \textbf{FR1.6:} Ghi nhận sự đồng ý của người dùng với điều khoản và
    chính sách bảo mật trước khi mở tài khoản (phiên bản, thời điểm); cho
    phép người dùng xem, tải về, sửa và gửi yêu cầu xóa dữ liệu cá nhân, xử
    lý yêu cầu xóa theo thời hạn lưu bắt buộc của dữ liệu giao dịch (NFR8.2,
    NFR8.5).
\end{itemize}

\subsubsection*{FR2: Quản lý hồ sơ nhà cung cấp}
\begin{itemize}
    \item \textbf{FR2.1:} Thiết lập, cập nhật hồ sơ nhà cung cấp (tên, ảnh,
    mô tả, vùng trồng/khu vực canh tác). Hồ sơ là trang nguồn gốc công khai,
    không phải gian hàng riêng của nhà cung cấp.
    \item \textbf{FR2.2:} Báo cáo thống kê cơ bản cho nhà cung cấp: doanh
    thu, số đơn hàng, sản phẩm bán chạy theo mốc thời gian.
    \item \textbf{FR2.3:} Nhà cung cấp theo dõi tồn kho của mình theo lô
    hàng (chi tiết tại FR13).
    \item \textbf{FR2.4:} Khai báo loại người bán cho nghĩa vụ thuế (tự sản
    xuất nông sản; hộ hoặc cá nhân kinh doanh; doanh nghiệp hoặc hợp tác xã)
    và nhóm thuế của sản phẩm; kiểm duyệt viên xác nhận khai báo cùng lúc với
    hồ sơ. Dữ liệu này dùng cho hóa đơn (FR7.2) và khấu trừ thuế thay (FR7.4).
\end{itemize}

\subsubsection*{FR3: Quản lý chứng nhận chất lượng}
\begin{itemize}
    \item \textbf{FR3.1:} Tải lên chứng nhận thuộc một trong các loại được
    hỗ trợ --- VietGAP, GlobalGAP, hữu cơ (organic), OCOP, HACCP --- kèm tệp
    minh chứng, mã số chứng nhận, cơ quan cấp và ngày hết hạn.
    \item \textbf{FR3.2:} Kiểm duyệt viên phê duyệt hoặc từ chối
    chứng nhận trước khi hiển thị công khai.
    \item \textbf{FR3.3:} Theo dõi hạn hiệu lực chứng nhận, tự động đánh dấu
    hết hạn và nhắc người bán gia hạn (liên hệ FR11.2).
\end{itemize}

\subsubsection*{FR4: Quản lý sản phẩm \& truy xuất nguồn gốc}
\begin{itemize}
    \item \textbf{FR4.1:} Đăng bán sản phẩm kèm hình ảnh, mô tả, đơn vị
    tính (kg, tấn, thùng), giá bán lẻ và bảng giá bán sỉ theo bậc số lượng.
    \item \textbf{FR4.2:} Ghi nhật ký canh tác theo lô (ngày gieo trồng, xử
    lý, thu hoạch). Mỗi lô được duyệt có một mã QR truy xuất duy nhất do hệ
    thống tự sinh; nhà cung cấp tải và in được mã này, và bất kỳ ai quét mã
    đều xem được trang truy xuất mà không cần đăng nhập.
    \item \textbf{FR4.3:} Giữ chỗ và trừ tồn trên đúng dòng tồn khi phát
    sinh giao dịch, tránh bán vượt số lượng thực có.
    \item \textbf{FR4.4:} Phân loại sản phẩm theo danh mục đa cấp chuyên
    ngành nông sản.
    \item \textbf{FR4.5:} Tìm kiếm và lọc nâng cao (khoảng giá, vùng miền,
    loại chứng nhận, đánh giá, tình trạng còn hàng); phía từ khóa chấp nhận
    gõ không dấu và gõ sai chính tả, kết hợp với tìm kiếm theo ngữ nghĩa
    (FR12.3).
    \item \textbf{FR4.6:} Ràng buộc tính hợp lệ của sản phẩm khi đăng bán:
    bắt buộc gắn danh mục lá thuộc cây danh mục nông sản đóng, kiểm tra đơn
    vị tính và các trường bắt buộc theo danh mục, đối chiếu với danh sách
    nhóm hàng bị loại trừ (mục~\ref{sec:quytrinhnghiepvu}).
    \item \textbf{FR4.7:} Tiếp nhận báo cáo từ người mua về sản phẩm sai
    danh mục hoặc không thuộc nhóm nông sản, đưa vào hàng đợi xử lý của
    kiểm duyệt viên (liên hệ FR10.2).
\end{itemize}

\subsubsection*{FR5: Giao dịch bán lẻ (B2C)}
\begin{itemize}
    \item \textbf{FR5.1:} Giỏ hàng gồm sản phẩm từ gian hàng bán lẻ của
    Farmery --- bên bán duy nhất ở luồng này.
    \item \textbf{FR5.2:} Đặt hàng: chọn địa chỉ, khung giờ giao và phương
    thức thanh toán; hiển thị đủ tiền hàng, thuế, cước giao trước khi xác nhận;
    giữ chỗ tồn cho đơn chờ thanh toán và tự nhả khi đơn quá hạn hoặc bị hủy;
    đơn sau giờ chốt đơn chuyển sang chuyến trung chuyển kế tiếp.
    \item \textbf{FR5.3:} Áp dụng mã giảm giá của gian hàng bán lẻ.
    \item \textbf{FR5.4:} Theo dõi đơn trên dòng thời gian tám trạng thái
    (Đã đặt, Đã thanh toán, Đang đóng gói, Lên chuyến trung chuyển, 3PL đã
    nhận, Đang giao, Đã giao, Hoàn tất hoặc Khiếu nại); hủy đơn trước khi đóng
    gói. Farmery chịu trách nhiệm trực tiếp với khách lẻ.
    \item \textbf{FR5.5:} Đánh giá sản phẩm (điểm số, nhận xét, hình ảnh)
    sau khi đơn hoàn tất; điểm đánh giá được tổng hợp vào uy tín của nhà cung
    cấp gốc; Farmery có thể phản hồi công khai.
    \item \textbf{FR5.6:} Mua lại: tạo giỏ hàng mới từ các sản phẩm của một
    đơn cũ, chỉ lấy những sản phẩm còn bán.
    \item \textbf{FR5.7:} Đăng ký nhận thông báo khi sản phẩm có hàng trở lại
    hoặc vào mùa; thông báo gửi qua FR11.1.
\end{itemize}

\subsubsection*{FR6: Giao dịch bán sỉ (B2B)}
\begin{itemize}
    \item \textbf{FR6.1:} Doanh nghiệp thu mua gửi yêu cầu báo giá (RFQ) cho
    một hoặc nhiều nông dân/HTX, nêu rõ số lượng, thời gian giao, tiêu chuẩn
    chất lượng mong muốn.
    \item \textbf{FR6.2:} Nông dân/HTX phản hồi RFQ (báo giá, số lượng có
    thể cung ứng); hai bên trao đổi, đàm phán trong kênh gắn với chính RFQ đó
    trước khi chốt.
    \item \textbf{FR6.3:} Khi hai bên chốt RFQ, hệ thống sinh hợp đồng từ
    mẫu chuẩn và lưu lịch sử phiên bản. Mỗi bên \emph{xác nhận giao kết} trong
    hệ thống bằng mã xác nhận dùng một lần; hệ thống lưu tài khoản, thời điểm
    và mã băm của đúng văn bản đã xác nhận để về sau chứng minh được nội dung
    hai bên đã đồng ý. Đơn sỉ được giao trong một đợt và thanh toán qua cổng
    thanh toán khi bên mua xác nhận nhận hàng.
    \item \textbf{FR6.4:} Tự động áp khung giá sỉ khi đạt số lượng tối
    thiểu (MOQ) đã thỏa thuận trong hợp đồng.
    \item \textbf{FR6.5:} Quản lý lịch giao hàng theo đợt (nhiều lần giao
    trên cùng một hợp đồng), theo dõi tiến độ và thanh toán của từng đợt
    \emph{(chỉ thiết kế)}.
\end{itemize}

\subsubsection*{FR7: Thanh toán, hóa đơn và thuế}
\begin{itemize}
    \item \textbf{FR7.1:} Hỗ trợ chuyển khoản ngân hàng/VietQR, ví điện tử
    qua cổng thanh toán trung gian đã được cấp phép, và COD có điều kiện: chỉ
    cho đơn dưới ngưỡng cấu hình và tài khoản chưa từng từ chối nhận hàng.
    Hoàn tiền đi qua cổng thanh toán theo quyết định xử lý khiếu nại.
    \item \textbf{FR7.2:} Phát hành hóa đơn điện tử cho đơn lẻ khi đơn hoàn
    tất, kèm mã tra cứu. Bản hiện thực phát hành hóa đơn giả lập (tệp PDF và mã
    tra cứu) qua một lớp tích hợp; kết nối nhà cung cấp dịch vụ hóa đơn điện
    tử thật \emph{(chỉ thiết kế)}.
    \item \textbf{FR7.3:} Ký quỹ trung gian (escrow) cho đơn sỉ có giá trị
    vượt ngưỡng: tiền của bên mua được giữ tại cổng thanh toán trung gian được
    cấp phép, chỉ giải ngân cho nhà cung cấp sau khi bên mua xác nhận nhận
    đủ hàng --- hệ thống không trực tiếp giữ tiền (liên hệ NFR3.4)
    \emph{(chỉ thiết kế)}.
    \item \textbf{FR7.4:} Đối soát đơn sỉ: tính hoa hồng theo biểu phí công
    bố, Farmery xuất hóa đơn dịch vụ cho nhà cung cấp; với nhà cung cấp là hộ
    hoặc cá nhân kinh doanh, khấu trừ và nộp thuế thay theo tỷ lệ của pháp
    luật thuế, cấp chứng từ khấu trừ, trả lại phần đã khấu trừ khi đơn bị hủy
    hoặc trả hàng \emph{(chỉ thiết kế: có sẵn trường dữ liệu đối soát, bản
    hiện thực không tính thuế)}.
\end{itemize}

\subsubsection*{FR8: Giao hàng \& khiếu nại}
\begin{itemize}
    \item \textbf{FR8.1:} Giao đơn lẻ qua ba chặng: nhà cung cấp tự đưa
    hàng về kho (chặng đầu; Farmery thuê xe gom \emph{(chỉ thiết kế)}); nhân
    viên kho lập \emph{chuyến trung chuyển} xe lạnh Lâm Đồng--TP.HCM từ danh
    sách đơn sau giờ chốt và cập nhật trạng thái chuyến (chặng giữa); tạo vận
    đơn với đơn vị vận chuyển nội thành theo khung giờ khách chọn, nhận trạng
    thái qua webhook (chặng cuối).
    \item \textbf{FR8.2:} Tính phí giao đơn lẻ gồm cước 3PL nội thành và phần
    bù chặng giữa; miễn phí khi đơn vượt ngưỡng cấu hình; hiển thị trước khi
    xác nhận đơn.
    \item \textbf{FR8.3:} Yêu cầu đổi trả, hoàn tiền và khiếu nại đơn hàng
    theo chính sách: thời hạn khiếu nại theo nhóm hàng, minh chứng ảnh hoặc
    video, hoàn tiền không cần trả hàng dưới ngưỡng cấu hình (ghi bút toán hủy
    vào sổ nhập xuất); nhân viên vận hành ra quyết định trong thời hạn cam kết
    (liên hệ FR10.3).
\end{itemize}

\subsubsection*{FR9: Tương tác người dùng}
\begin{itemize}
    \item \textbf{FR9.1:} Trao đổi bất đồng bộ gắn với một đơn hàng hoặc một
    yêu cầu khiếu nại, giữa khách và nhân viên vận hành, có đính kèm ảnh.
    Không có kênh nhắn tin tự do giữa người dùng; đàm phán sỉ dùng kênh của
    FR6.2.
    \item \textbf{FR9.2:} Danh sách sản phẩm yêu thích.
\end{itemize}

\subsubsection*{FR10: Vận hành và quản trị nền tảng}
\begin{itemize}
    \item \textbf{FR10.1:} Xét duyệt hồ sơ nhà cung cấp, kiểm duyệt định kỳ
    sản phẩm và chứng nhận (liên hệ FR3.2), khóa/mở tài khoản vi phạm.
    \item \textbf{FR10.2:} Giám sát vi phạm (sản phẩm sai thông tin, đánh
    giá giả, gian lận đơn hàng).
    \item \textbf{FR10.3:} Tiếp nhận và xử lý khiếu nại, đổi trả theo hàng
    đợi có hạn xử lý; tự động nâng mức ưu tiên khi quá hạn; đóng vai trò
    trung gian phân xử khi có tranh chấp giữa các bên.
    \item \textbf{FR10.4:} Báo cáo thống kê tổng quan hệ thống: tổng giá trị
    giao dịch, số lượng người dùng hoạt động, tỷ lệ hoàn tất đơn hàng.
    \item \textbf{FR10.5:} Bảng điều khiển giám sát tồn kho và giao hàng:
    lô sắp hết hạn hoặc tồn thấp (liên hệ FR2.3, FR11.2), chuyến trung chuyển
    trong ngày, đơn trễ và sự cố với đơn vị vận chuyển (FR8.1).
    \item \textbf{FR10.6:} Cấu hình nền tảng: cây danh mục nông sản, ngưỡng
    của các luật giám sát, biểu phí công bố, thời hạn khiếu nại theo nhóm
    hàng, ngưỡng hoàn tiền không cần trả hàng, điều kiện COD, giờ chốt đơn,
    khung giờ giao, ngưỡng miễn phí giao; tạo tài khoản nội bộ và gán vai trò.
    \item \textbf{FR10.7:} Ghi nhật ký mọi thao tác của nhân sự nội bộ có tác
    động tới dữ liệu của người dùng --- duyệt, khóa, phân xử, thu hồi lô, và
    nhập liệu thay nhà cung cấp --- kèm người thực hiện, thời điểm và lý do;
    nhà cung cấp xem được các thao tác đã thực hiện thay mình.
\end{itemize}

\noindent FR10.1 và FR10.2 thuộc kiểm duyệt viên; FR10.3 đến FR10.5 thuộc
nhân viên vận hành; FR10.6 thuộc quản trị viên hệ thống; FR10.7 áp dụng cho
mọi vai trò nội bộ (Bảng~\ref{tab:vaitro-noibo}).

\subsubsection*{FR11: Thông báo}
\begin{itemize}
    \item \textbf{FR11.1:} Thông báo đa kênh (trên giao diện web, thông báo
    đẩy của ứng dụng web cài đặt được, email; SMS \emph{(chỉ thiết kế)}) ở mỗi lần đơn chuyển trạng
    thái, khi khiếu nại có phản hồi, khi sản phẩm đã đăng ký có hàng, và về
    tiến độ RFQ, hợp đồng.
    \item \textbf{FR11.2:} Cảnh báo chứng nhận sắp hết hạn (liên hệ FR3.3)
    và tồn kho chạm ngưỡng tối thiểu (liên hệ FR2.3).
\end{itemize}

\subsubsection*{FR12: Hỗ trợ AI}
\label{subsec:FR-AI}
Chi tiết thiết kế tại mục~\ref{sec:thietkeAI}.
\begin{itemize}
    \item \textbf{FR12.1:} Gợi ý sản phẩm tương tự ở trang chi tiết sản phẩm,
    dựa trên độ gần của vector mô tả sản phẩm. Gợi ý cá nhân hóa theo lịch sử
    tìm kiếm, mua hàng là bước sau khi có đủ dữ liệu tương tác.
    \item \textbf{FR12.2:} Dự báo xu hướng giá theo mùa vụ, hiển thị cho nhân
    viên thu mua ngay tại màn hình lập đơn thu mua và đặt giá bán lẻ, làm căn
    cứ tham khảo.
    \item \textbf{FR12.3:} Tìm kiếm theo ngữ nghĩa: hiểu câu tìm kiếm tự
    nhiên (ví dụ ``rau ăn lẩu'', ``trái cây ít ngọt'') để trả về sản phẩm phù
    hợp dù mô tả không chứa đúng từ khóa. Tìm nhà cung cấp theo ngữ nghĩa cho
    khách sỉ \emph{(chỉ thiết kế)}.
\end{itemize}

\subsubsection*{FR13: Thu mua và quản lý kho (luồng 1P)}
\begin{itemize}
    \item \textbf{FR13.1:} Nhân viên thu mua lập đơn thu mua từ chào hàng của
    nhà cung cấp; đơn chỉ có hiệu lực khi nhà cung cấp xác nhận trên hệ thống,
    và từ đó phần sản lượng trong đơn được giữ chỗ.
    \item \textbf{FR13.2:} Nhân viên kho nghiệm thu chất lượng, ghi nhận phần
    đạt, phần bị từ chối và hao hụt; phần đạt trở thành dòng tồn của Farmery,
    giữ nguyên mã lô, nhật ký canh tác và mã QR.
    \item \textbf{FR13.3:} Quản lý tồn kho theo dòng tồn (lô, chủ sở hữu,
    nơi chứa); mọi thay đổi số lượng đi qua sổ nhập xuất theo đúng nguồn giao
    dịch tại Bảng~\ref{tab:nguon-giaodich} (Phụ lục~Q), không sửa trực tiếp con số tồn.
    \item \textbf{FR13.4:} Đóng gói hàng nhập kho thành đơn vị bán lẻ trong
    một bước, in tem gồm mã QR của lô và nội dung nhãn hàng hóa bắt buộc; ghi
    hao hụt khi đóng gói. Sơ chế, phân loại nhiều bước và kiểm kê định kỳ
    \emph{(chỉ thiết kế)}.
    \item \textbf{FR13.6:} Thu hồi lô do nhân viên vận hành thực hiện: khóa
    mọi dòng tồn của lô ở cả hai luồng, lập danh sách các đơn đã bán từ lô
    đó, thông báo cho người mua, và hiển thị cảnh báo thu hồi trên trang truy
    xuất của lô.
\end{itemize}

\subsection{Yêu cầu phi chức năng}
\label{sec:yeucauphichucnang}

Các yêu cầu phi chức năng dưới đây cũng tuân theo nguyên tắc đã nêu ở đầu mục~\ref{sec:yeucau}: mỗi ngưỡng đều phải truy được về một nhu cầu hoặc khó khăn cụ thể của người dùng, chứ không lấy từ thông lệ kỹ thuật chung. Ví dụ ngưỡng thời gian phản hồi không xuất phát từ ``thông lệ duyệt web'' mà từ việc nhà cung cấp thao tác trên mạng 4G không ổn định tại vùng trồng; ngưỡng khả dụng không xuất phát từ tài liệu kiểm thử mà từ điểm đau ``ngại biểu mẫu phức tạp'' của chính nhóm này. Cột \emph{Căn cứ} trong Bảng~\ref{tab:nguongNFR} (Phụ lục L) ghi rõ nhu cầu gốc của từng ngưỡng.

\subsubsection*{NFR1: Hiệu năng}
\begin{itemize}
    \item \textbf{NFR1.1:} Thời gian tải trang chủ và trả kết quả tìm
    kiếm/lọc sản phẩm ở mức chấp nhận được; ảnh sản phẩm được phục vụ ở định
    dạng WebP với nhiều kích cỡ để không làm chậm lần hiển thị nội dung chính;
    số đo thực tế đối chiếu tại mục~\ref{sec:danhgiahieunang}.
    \item \textbf{NFR1.2:} Đảm bảo không xảy ra bán vượt tồn kho (oversell)
    khi nhiều giao dịch thao tác đồng thời trên cùng một dòng tồn --- nhiều đơn bán lẻ, hoặc hợp đồng bán sỉ cùng lúc với đơn thu mua.
\end{itemize}

\subsubsection*{NFR2: Khả năng mở rộng}
\begin{itemize}
    \item \textbf{NFR2.1:} Kiến trúc cho phép bổ sung vai trò/mô-đun mới
    (ví dụ thêm một vai trò nội bộ ngoài sáu vai trò đã thiết kế tại
    Bảng~\ref{tab:vaitro-noibo}, hoặc mở thêm kênh bán mới) mà không
    thay đổi lớn hệ thống hiện có. Việc một tài khoản có thể mang nhiều vai
    trò, và mỗi vai trò chỉ là một tập quyền độc lập, là cơ sở để bổ sung vai
    trò mà không phải sửa các vai trò sẵn có.
    \item \textbf{NFR2.2:} Các thành phần xử lý tải cao (tìm kiếm, đặt
    hàng) được thiết kế để có thể mở rộng theo chiều ngang khi lưu lượng
    tăng.
\end{itemize}

\subsubsection*{NFR3: Bảo mật}
\begin{itemize}
    \item \textbf{NFR3.1:} Phân quyền theo RBAC (Chương~\ref{chap:coso}), tuân
    thủ nguyên tắc đặc quyền tối thiểu.
    \item \textbf{NFR3.2:} Mã hóa dữ liệu nhạy cảm khi lưu trữ (thông tin
    định danh, MST, tài khoản ngân hàng nếu có), kể cả các tệp riêng tư như
    ảnh giấy tờ eKYC, giấy phép kinh doanh và hợp đồng; tệp riêng tư chỉ đọc
    được qua đường dẫn có chữ ký, hết hạn sau thời gian ngắn.
    \item \textbf{NFR3.3:} Băm mật khẩu bằng thuật toán an toàn
    (Bcrypt/Argon2).
    \item \textbf{NFR3.4:} Nghiệp vụ thanh toán ủy quyền xử lý qua cổng
    thanh toán trung gian đã đạt chuẩn bảo mật thẻ (PCI-DSS do bên cổng
    thanh toán chịu trách nhiệm), hệ thống không lưu trữ trực tiếp dữ liệu
    thẻ.
    \item \textbf{NFR3.5:} Giới hạn tần suất thao tác (rate limiting) đối
    với các hành vi có nguy cơ gian lận (đặt đơn ảo, đăng nhập sai nhiều
    lần).
    \item \textbf{NFR3.6:} Ứng dụng nội bộ bắt buộc xác thực hai lớp cho mọi
    tài khoản nhân sự, phiên làm việc ngắn, có thể giới hạn theo địa chỉ mạng;
    các đường dẫn API nội bộ tách tiền tố và có lớp kiểm tra quyền riêng.
\end{itemize}

\subsubsection*{NFR4: Độ tin cậy \& tính sẵn sàng}
\begin{itemize}
    \item \textbf{NFR4.1:} Các thao tác ảnh hưởng số dư/tồn kho (đặt hàng,
    hủy đơn, hoàn tiền) đảm bảo tính nhất quán dữ liệu trong phạm vi một
    giao dịch cơ sở dữ liệu.
    \item \textbf{NFR4.2:} Có cơ chế sao lưu định kỳ dữ liệu hệ thống.
\end{itemize}

\subsubsection*{NFR5: Khả năng bảo trì}
\begin{itemize}
    \item \textbf{NFR5.1:} Tổ chức mã nguồn theo module tách biệt theo
    nghiệp vụ (tài khoản, sản phẩm, đơn hàng, thanh toán) để dễ bảo trì và
    mở rộng.
    \item \textbf{NFR5.2:} Tài liệu hóa API theo chuẩn OpenAPI/Swagger cho
    các endpoint chính.
\end{itemize}

\subsubsection*{NFR6: Khả dụng cho người dùng ít rành công nghệ}
\begin{itemize}
    \item \textbf{NFR6.1:} Giao diện đơn giản, hướng dẫn rõ ràng cho nhóm
    nông dân/HTX (mục~\ref{sec:doituongnguoidung}), ưu tiên hiển thị tốt
    trên thiết bị di động.
    \item \textbf{NFR6.2:} Có nhân viên hỗ trợ vùng hướng dẫn trực tiếp và nhập
    liệu hộ khi cần, để giảm rào cản thao tác cho người dùng ít quen thuộc với
    công nghệ.
    \item \textbf{NFR6.3:} Luồng đăng bán sản phẩm và ghi nhật ký canh tác
    được thiết kế tối giản, giảm số bước nhập liệu.
\end{itemize}

\subsubsection*{NFR7: Khả năng tích hợp \& tương thích}
\begin{itemize}
    \item \textbf{NFR7.1:} Tích hợp qua API chuẩn (RESTful) với đơn vị vận
    chuyển và cổng thanh toán trung gian.
    \item \textbf{NFR7.2:} Giao diện web đáp ứng (responsive), hoạt động
    tốt trên các trình duyệt phổ biến.
    \item \textbf{NFR7.3:} Ứng dụng mua hàng là ứng dụng web cài đặt được
    (PWA): cài được lên màn hình chính của điện thoại, nhận được thông báo
    đẩy, và thao tác ghi nhật ký canh tác của nhà cung cấp tại ruộng vẫn thực
    hiện được khi mất kết nối, được đồng bộ khi có mạng trở lại. Ứng dụng nội
    bộ có giao diện đáp ứng, dùng được trên điện thoại cho nhân viên kho và
    nhân viên hỗ trợ vùng.
\end{itemize}

\subsubsection*{NFR8: Pháp lý \& tuân thủ}
\begin{itemize}
    \item \textbf{NFR8.1:} Tuân thủ Luật Thương mại điện tử 2025 và nghị
    định hướng dẫn với cả hai vai trò của Farmery: công khai thông tin chủ
    quản và các chính sách (bảo mật, giá và phí, thanh toán, giao hàng, đổi
    trả--hoàn tiền, khiếu nại, tiêu chí ưu tiên hiển thị) ở vị trí dễ thấy;
    người dùng đồng ý trước khi mở tài khoản; xác thực danh tính người bán
    trước khi cho bán; kiểm duyệt nội dung trước khi hiển thị; công khai nội
    dung nhãn bắt buộc của sản phẩm.
    \item \textbf{NFR8.2:} Tuân thủ Luật Bảo vệ dữ liệu cá nhân 2025 và
    nghị định hướng dẫn: thu thập có đồng ý, đúng mục đích, tối thiểu; chủ
    thể dữ liệu xem, sửa, yêu cầu xóa được (FR1.6).
    \item \textbf{NFR8.3:} Hóa đơn điện tử (FR7.2) đúng nội dung và thời
    điểm lập theo quy định về hóa đơn, chứng từ. Bản hiện thực dùng hóa đơn
    giả lập; tuân thủ đầy đủ được kiểm tra khi kết nối nhà cung cấp dịch vụ
    thật.
    \item \textbf{NFR8.4:} Khấu trừ, nộp thuế thay cho nhà cung cấp là hộ
    hoặc cá nhân kinh doanh trên sàn sỉ theo quy định quản lý thuế với hoạt
    động thương mại điện tử (FR7.4) \emph{(chỉ thiết kế)}.
    \item \textbf{NFR8.5:} Lưu trữ và bảo đảm truy cập được dữ liệu liên
    quan đến hợp đồng đã giao kết ít nhất 3 năm, và dữ liệu nhãn hàng hóa,
    chứng từ, hóa đơn kèm dữ liệu giao dịch ít nhất 5 năm; trong thời hạn này
    yêu cầu xóa dữ liệu cá nhân chỉ được thực hiện bằng ẩn danh hóa phần không
    bắt buộc lưu.
\end{itemize}

\subsubsection*{NFR9: Toàn vẹn dữ liệu truy xuất nguồn gốc}
\begin{itemize}
    \item \textbf{NFR9.1:} Dữ liệu nhật ký canh tác và chứng nhận, một khi
    đã được duyệt và công khai, cần được kiểm soát chặt chẽ trước khi cho
    phép chỉnh sửa nhằm đảm bảo tính bất biến (immutability) của thông tin
    truy xuất --- liên hệ mục~\ref{subsec:gs1blockchain}.
\end{itemize}

\noindent\textbf{Mức độ ưu tiên và cách kiểm thử.} Chín nhóm yêu cầu phi chức năng cũng được đánh giá theo cùng ba thuộc tính như phần yêu cầu chức năng. Bảy nhóm ở mức Must --- trong đó hai nhóm khó nhất về mặt kỹ thuật là chống bán vượt tồn kho khi giao dịch đồng thời, và chịu tải ở mức 100 người dùng --- còn hai nhóm ở mức Should là khả năng mở rộng vai trò và độ khả dụng cho nhóm nhà cung cấp. Ma trận đầy đủ kèm công cụ kiểm thử cho từng nhóm được trình bày tại Phụ lục M.

\subsubsection*{Ngưỡng đo lường và căn cứ}
\label{subsec:nguongNFR}

Các yêu cầu phi chức năng ở trên mô tả \textit{điều hệ thống cần đạt}; việc
\textit{đo bằng chỉ số nào, ngưỡng bao nhiêu và vì sao chọn ngưỡng đó} được
tách riêng thành Bảng~\ref{tab:nguongNFR} tại Phụ lục L, dùng làm căn cứ đối

cầu mang tính định tính hoặc tuân thủ pháp lý thuần túy (NFR3.4, NFR5.1,
NFR8.1--NFR8.5) được nghiệm thu bằng checklist thay vì chỉ số đo, nên
không xuất hiện trong bảng đó.


\subsection{Yêu cầu dữ liệu}
\label{subsec:yeucaudulieu}

Hệ thống sàn thương mại điện tử nông nghiệp Farmery vận hành theo mô hình B2B2C, đòi hỏi xử lý song song các luồng dữ liệu thương mại tức thời, dữ liệu chuỗi cung ứng bất biến và dữ liệu phục vụ phân tích/học máy. Mục này xác lập các yêu cầu dữ liệu chuẩn tắc làm cơ sở trực tiếp cho thiết kế cơ sở dữ liệu chi tiết tại mục~\ref{sec:csdl}.

\subsubsection*{1. Phân loại các miền dữ liệu (Data Domains)}
\begin{itemize}
    \item \textbf{Dữ liệu định danh và phân quyền (IAM):} thông tin tài khoản, mã băm mật khẩu, thông tin liên lạc đã xác thực OTP, vai trò RBAC, hồ sơ doanh nghiệp (mã số thuế, giấy phép kinh doanh) và hồ sơ eKYC của nhà cung cấp.
    \item \textbf{Dữ liệu danh mục và sản phẩm:} cây danh mục nông sản đa cấp đóng, đơn vị tính hợp lệ theo danh mục, giá bán lẻ, bảng giá sỉ theo bậc sản lượng, thông tin vùng trồng và mùa vụ.
    \item \textbf{Dữ liệu truy xuất nguồn gốc và chứng nhận:} định danh lô sản xuất, nhật ký canh tác theo mốc thời gian (gieo giống, xử lý phân bón và thuốc bảo vệ thực vật, thu hoạch), hồ sơ chứng nhận chất lượng kèm cơ quan cấp và thời hạn hiệu lực, mã QR định danh theo nguyên lý GS1.
    \item \textbf{Dữ liệu thu mua (luồng 1P):} chào hàng của nhà cung cấp, đơn thu mua theo lô, kết quả nghiệm thu chất lượng khi nhập kho, giá vốn và tỷ lệ hao hụt thực tế theo từng lô.
    \item \textbf{Dữ liệu giao dịch thương mại:} giỏ hàng phiên và đơn bán lẻ; yêu cầu báo giá (RFQ), lịch sử đàm phán, hợp đồng điện tử kèm mã băm và biên bản giao nhận ở luồng bán sỉ; giao dịch thanh toán, hoàn tiền, hóa đơn và dữ liệu đối soát hoa hồng.
    \item \textbf{Dữ liệu vận chuyển và tồn kho:} sổ địa chỉ giao nhận, chuyến trung chuyển kèm danh sách đơn, vận đơn tích hợp đơn vị vận chuyển nội thành, lịch sử trạng thái hành trình, tồn kho theo lô kèm hạn sử dụng theo nguyên tắc FEFO và phiếu khiếu nại chất lượng.
    \item \textbf{Dữ liệu uy tín và tương tác:} điểm đánh giá, nội dung nhận xét, hình ảnh thực tế khi nhận hàng, điểm uy tín tổng hợp của nhà cung cấp tính theo trọng số thời gian, luồng trao đổi gắn với đơn hàng và khiếu nại.
    \item \textbf{Dữ liệu tệp:} ảnh sản phẩm, ảnh nhật ký canh tác, minh chứng chứng nhận và khiếu nại, ảnh giấy tờ eKYC, hợp đồng và hóa đơn; mỗi tệp được phân loại công khai hoặc riêng tư, kèm mã băm và thời hạn lưu.
    \item \textbf{Dữ liệu tuân thủ:} lần đồng ý với điều khoản và chính sách bảo mật (phiên bản, thời điểm), yêu cầu của chủ thể dữ liệu, loại người bán và nhóm thuế của sản phẩm.
    \item \textbf{Dữ liệu phân tích và học máy:} nhật ký sự kiện hành vi (clickstream, từ khóa tìm kiếm, lượt thêm giỏ); chuỗi time series lịch sử giá và sản lượng theo vùng; cảnh báo do luật ngưỡng sinh ra kèm phán quyết của kiểm duyệt viên hoặc nhân viên vận hành, tích lũy làm nhãn cho giai đoạn sau.
    \item \textbf{Dữ liệu quản trị và lưu vết:} nhật ký thao tác quản trị, hồ sơ vi phạm và quyết định xử lý kèm lý do.
\end{itemize}

\subsubsection*{2. Ràng buộc toàn vẹn và chất lượng dữ liệu}
\begin{itemize}
    \item \textbf{Ràng buộc nhất quán giao dịch (ACID):} Mọi thao tác biến động tồn kho (giữ chỗ, trừ tồn, nhả tồn) hoặc trạng thái tiền (thanh toán, hoàn tiền) bắt buộc phải thực thi trong cùng một transaction; số lượng tồn kho khả dụng không được phép mang giá trị âm ($AvailableQty \ge 0$).
    \item \textbf{Tính bất biến của dữ liệu truy xuất (Append-only):} Nhật ký canh tác và chứng chỉ chất lượng một khi đã được kiểm duyệt viên phê duyệt công khai không được phép sửa đổi hoặc xóa trực tiếp. Mọi bổ sung, đính chính phải được ghi nhận dưới dạng bản ghi mới (versioning) kèm định danh người sửa và lý do.
    \item \textbf{Toàn vẹn giá vốn và doanh thu (luồng 1P):} mỗi đơn bán lẻ phải truy được về đúng lô đã thu mua, để đối chiếu giá bán với giá vốn và tỷ lệ hao hụt thực tế của lô đó.
\end{itemize}

\subsubsection*{3. Bảo mật, quyền riêng tư và vòng đời dữ liệu}
\begin{itemize}
    \item \textbf{Tuân thủ pháp luật về bảo vệ dữ liệu cá nhân (NFR8.2):} Dữ liệu nhạy cảm (số CCCD, tài khoản ngân hàng, mã số thuế) phải được mã hóa ở mức lưu trữ (Encryption at Rest) bằng thuật toán AES-256. Mật khẩu người dùng được băm một chiều bằng Bcrypt (cost factor $\ge 10$).
    \item \textbf{Không lưu trữ thông tin thẻ thanh toán:} Tuyệt đối không lưu trữ số thẻ tín dụng/CVV trên hệ thống Farmery; toàn bộ luồng thanh toán ủy quyền qua cổng trung gian đạt chuẩn PCI-DSS (NFR3.4).
    \item \textbf{Vòng đời và sao lưu dữ liệu (Backup \& Archiving):} Thực hiện snapshot CSDL tự động định kỳ 24 giờ một lần (RPO $\le 24$h, RTO $\le 4$h). Dữ liệu giao dịch hoàn tất sau 1 năm được chuyển sang phân vùng lưu trữ lạnh (Cold storage) để tối ưu hiệu năng truy vấn cho bảng hoạt động, nhưng vẫn truy cập được trong suốt thời hạn lưu bắt buộc (NFR8.5).
\end{itemize}

\subsubsection*{4. Yêu cầu dữ liệu cho các mô-đun AI}
\begin{itemize}
    \item \textbf{Xử lý bài toán thiếu dữ liệu khởi đầu (Cold-start):} Gợi ý sản phẩm tương tự dựa trên vector mô tả và đặc trưng nông sản tĩnh (vùng miền, mùa vụ, chứng nhận OCOP/VietGAP) nên chạy được ngay khi chưa có lịch sử tương tác; lọc cộng tác (collaborative filtering) chỉ được bổ sung khi đã tích lũy đủ dữ liệu hành vi.
    \item \textbf{Chuẩn hóa chuỗi thời gian dự báo giá:} Dữ liệu giá nông sản cần được nội suy làm sạch giá trị khuyết (missing data), làm mịn dữ liệu đột biến do thời tiết cực đoan và gắn nhãn theo tuần/tháng.
\end{itemize}
Đặc tả chi tiết các thực thể và quan hệ được trình bày tại mục~\ref{sec:csdl}.

\section{Kết chương}

Chương này đã xác lập hệ thống đề xuất ở mức nghiệp vụ và yêu cầu: mô hình kinh doanh hai luồng cùng điều kiện hòa vốn của luồng bán lẻ, các quy trình nghiệp vụ từ thu mua, giao hàng ba chặng tới đổi trả kèm quy tắc và các chính sách pháp lý, đạo đức; tổng quan hệ thống với hai ứng dụng, ba nhóm người dùng bên ngoài và sáu vai trò nhân sự nội bộ; năm vấn đề nghiên cứu và ba thách thức chính; cùng bộ yêu cầu chức năng (FR1--FR13), phi chức năng có ngưỡng đo lường (NFR1--NFR9) và yêu cầu dữ liệu. Đây là đầu vào trực tiếp cho việc mô hình hóa và thiết kế hệ thống tại Chương~\ref{chap:phantich}.
