\chapter{Kiến thức và công nghệ nền tảng}
\label{chap:coso}

Chương này trình bày các kiến thức nền tảng mà Farmery vận dụng: đặc điểm của chính mặt hàng nông sản và hệ quả của chúng lên thiết kế hệ thống, bối cảnh chính sách và tiêu chuẩn chất lượng tại Việt Nam, các lý thuyết về mô hình thương mại điện tử, chuỗi cung ứng, truy xuất nguồn gốc, uy tín và công nghệ nền tảng, cùng các bài toán trí tuệ nhân tạo được tích hợp vào hệ thống. Các nền tảng và nghiên cứu cụ thể có liên quan được khảo sát riêng tại Chương~\ref{chap:lienquan}.

\section{Đặc điểm hàng nông sản}
\label{sec:dacdiemnongsan}

Toàn bộ thiết kế của Farmery xuất phát từ một điểm: nông sản không hành xử giống các loại hàng hóa khác trên sàn thương mại điện tử. Mục này làm rõ phạm vi khái niệm ``nông sản'' được dùng trong đồ án, các đặc điểm phân biệt nông sản với hàng hóa thông thường, và các chứng nhận mà một lô nông sản cần có để ra thị trường.

\subsection{Phạm vi khái niệm nông sản trong đồ án}
\label{subsec:dinhnghianongsan}

Pháp luật Việt Nam không có một định nghĩa duy nhất về ``nông sản'' áp dụng cho mọi mục đích. Đồ án xác định phạm vi dựa trên ba căn cứ:

\begin{itemize}
    \item \textbf{Luật An toàn thực phẩm số 55/2010/QH12} \parencite{luat_attp_2010} --- Điều 2 định nghĩa \emph{thực phẩm tươi sống} là ``thực phẩm chưa qua chế biến bao gồm thịt, trứng, cá, thuỷ hải sản, rau, củ, quả tươi và các thực phẩm khác chưa qua chế biến'' (khoản 21); phân biệt \emph{sơ chế} --- xử lý sản phẩm trồng trọt để tạo ra thực phẩm tươi sống có thể ăn ngay hoặc nguyên liệu cho khâu chế biến (khoản 16) --- với \emph{chế biến}, tức xử lý theo phương pháp công nghiệp hoặc thủ công để tạo thành sản phẩm thực phẩm (khoản 4).
    \item \textbf{Nghị định số 15/2018/NĐ-CP} \parencite{nd15_2018} --- Điều 4 khoản 1 quy định nhóm phải tự công bố sản phẩm, trong đó có \emph{thực phẩm đã qua chế biến bao gói sẵn}. Đây là ranh giới pháp lý phân biệt nhóm nông sản tươi hoặc sơ chế (không phải tự công bố) với nhóm chế biến sâu đóng gói (bắt buộc tự công bố).
    \item \textbf{Luật Trồng trọt số 31/2018/QH14} \parencite{luat_trongtrot_2018} --- xác định hoạt động trồng trọt bao gồm cả khâu thu hoạch, sơ chế, bảo quản, chế biến và thương mại sản phẩm cây trồng, cho thấy chuỗi giá trị nông sản được pháp luật nhìn nhận là liên tục từ ruộng tới thị trường.
\end{itemize}

Trên cơ sở đó, hàng hóa được chia thành bốn nhóm với ranh giới trong và ngoài phạm vi như sau:
\begin{itemize}
    \item \textbf{Nhóm 1 --- Nông sản tươi chưa qua chế biến} (rau, củ, quả, gạo, các loại hạt): \emph{trong phạm vi}, là trục chính của nền tảng.
    \item \textbf{Nhóm 2 --- Nông sản sơ chế hoặc bảo quản đơn giản} (sấy khô, cấp đông, đóng gói hút chân không, không thêm phụ gia): \emph{trong phạm vi}.
    \item \textbf{Nhóm 3 --- Nông sản chế biến sâu} (nước ép, mứt, trà, tinh dầu --- phần lớn sản phẩm OCOP thuộc nhóm này): \emph{trong phạm vi có điều kiện} --- nhà cung cấp bắt buộc phải có bản tự công bố sản phẩm và giấy chứng nhận cơ sở đủ điều kiện an toàn thực phẩm.
    \item \textbf{Nhóm 4 --- Ngoài phạm vi}: thủy sản sống, thịt tươi, động vật sống (đòi hỏi điều kiện kiểm dịch và bảo quản khác hẳn); thực phẩm chức năng, thuốc bảo vệ thực vật, vật tư nông nghiệp và mọi hàng hóa phi thực phẩm.
\end{itemize}

Ranh giới này không chỉ mang tính học thuật: nó là căn cứ trực tiếp để xây dựng danh mục sản phẩm đóng và danh sách loại trừ ở khâu đăng bán, nhằm ngăn nhà cung cấp đăng nhầm hoặc cố tình đăng sai loại hàng (mục~\ref{sec:quytrinhnghiepvu}).

\subsection{Nông sản khác hàng hóa thông thường ở điểm nào}
\label{subsec:sosanhnongsan}

So với hàng công nghiệp và hàng tiêu dùng đóng gói --- nhóm hàng mà các sàn thương mại điện tử phổ thông được thiết kế để phục vụ --- nông sản có bốn khác biệt chi phối toàn bộ thiết kế hệ thống:

\begin{itemize}
    \item \textbf{Mau hỏng, hạn sử dụng tính bằng ngày.} Hàng công nghiệp tồn kho hàng năm không mất giá trị; rau quả tươi mất giá trị theo từng ngày. Hệ quả: tồn kho phải quản lý theo lô kèm hạn sử dụng, ưu tiên xuất trước lô hết hạn trước, và thời gian giao hàng trở thành ràng buộc chứ không phải tiện ích. Đi kèm là tỷ lệ tổn thất sau thu hoạch 20--40\% (Chương~\ref{chap:gioithieu}), buộc phải tính chi phí hao hụt vào mô hình kinh doanh.
    \item \textbf{Nguồn cung theo mùa vụ, không theo đơn đặt hàng.} Nhà máy sản xuất thêm khi có cầu; nông dân không thể thu hoạch sớm hơn mùa vụ. Hệ quả: giá và sản lượng biến động mạnh theo thời điểm trong năm, cần dự báo thay vì chỉ ghi nhận.
    \item \textbf{Chất lượng không đồng nhất giữa các lô.} Hai lô cùng giống, cùng vùng trồng vẫn khác nhau về kích cỡ và độ chín, lại bán theo khối lượng hoặc theo bó chứ không theo ``cái''. Hệ quả: dữ liệu mô tả phải gắn theo từng lô thay vì theo mã sản phẩm, và hệ thống phải hỗ trợ nhiều đơn vị tính.
    \item \textbf{Người mua không tự kiểm chứng được chất lượng.} Người mua không thể nhìn quả cà chua để biết nó có dư lượng thuốc bảo vệ thực vật hay không --- kể cả sau khi đã ăn. Đây chính là loại hàng hóa mà lý thuyết bất cân xứng thông tin (mục~\ref{subsec:reputation}) mô tả. Cộng với ràng buộc pháp lý về an toàn thực phẩm, truy xuất nguồn gốc và chứng nhận bên thứ ba trở thành bắt buộc chứ không phải tùy chọn, và khâu kiểm duyệt không thể chỉ dựa vào tự động hóa.
\end{itemize}

Phụ lục G đối chiếu đầy đủ chín tiêu chí giữa nông sản, hàng công nghiệp và hàng tiêu dùng đóng gói, kèm ràng buộc thiết kế tương ứng của từng tiêu chí.

\subsection{Chứng nhận và giấy tờ để nông sản ra thị trường}
\label{subsec:chungnhan}

Một lô nông sản muốn đi từ trang trại ra thị trường --- đặc biệt là thị trường xuất khẩu --- phải kèm theo hai loại hồ sơ.

\noindent\textbf{Chứng nhận về cách thức sản xuất}, cấp cho cơ sở và có hiệu lực nhiều năm, chia thành ba nhóm. Nhóm \emph{thực hành nông nghiệp tốt} gồm GlobalG.A.P. --- chuẩn được công nhận rộng rãi nhất, bao phủ từ trước thu hoạch tới đóng gói, có thể bổ sung mô-đun GRASP về điều kiện lao động \parencite{globalgap_ifa} --- và VietGAP cho chuỗi cung ứng nội địa. Nhóm \emph{canh tác hữu cơ} gồm USDA Organic \parencite{usda_nop}, EU Organic \parencite{eu_organic_2018848} và JAS Organic. Nhóm \emph{an toàn thực phẩm cho khâu chế biến} lấy HACCP làm nền, phát triển lên thành ISO 22000, FSSC 22000, BRCGS Food Safety và IFS Food --- trong đó bốn chuẩn sau nằm trong danh sách được Sáng kiến An toàn Thực phẩm Toàn cầu công nhận \parencite{gfsi_cpo}. Ngoài ra còn các chứng nhận phục vụ thị trường ngách như Halal, Kosher, Fairtrade và Rainforest Alliance.

\noindent\textbf{Giấy tờ đi theo từng lô hàng}, cấp riêng cho mỗi chuyến hàng. Quan trọng nhất là giấy chứng nhận kiểm dịch thực vật lập theo chuẩn ISPM 12 \parencite{ippc_ispm12}, giấy chứng nhận xuất xứ để hưởng ưu đãi thuế quan, và phiếu kết quả phân tích dư lượng. Để xuất khẩu chính ngạch còn cần mã số vùng trồng và mã số cơ sở đóng gói, tại Việt Nam cấp theo Nghị định số 38/2026/NĐ-CP \parencite{ppd_masovungtrong}. Ở thị trường nội địa, nhóm chế biến sâu phải có giấy chứng nhận cơ sở đủ điều kiện an toàn thực phẩm và bản tự công bố sản phẩm.

Phụ lục H liệt kê đầy đủ từng chứng nhận kèm phạm vi áp dụng, thị trường đích và mức độ hỗ trợ của Farmery. Trong phạm vi đồ án, hệ thống hỗ trợ tải lên và kiểm duyệt nhóm chứng nhận về cách thức sản xuất; nhóm giấy tờ theo lô phục vụ xuất khẩu nằm ngoài phạm vi (mục~\ref{sec:phamvi}) nhưng được nêu ở đây để thấy bức tranh đầy đủ mà một nền tảng nông sản hoàn chỉnh phải xử lý.

\section{Bối cảnh chính sách và tiêu chuẩn chất lượng nông sản}
\label{sec:boicanhchinhsach}

Chương trình OCOP (Mỗi xã một sản phẩm) là chương trình quốc gia hỗ trợ chuẩn hóa và nâng cao giá trị sản phẩm nông thôn/nông sản đặc trưng vùng miền. Hình~\ref{fig:ocop} tổng hợp số liệu tăng trưởng từ 2024 đến 2025 \parencite{danviet_2024_ocop, vietnamnet_2025_ocop}. Giai đoạn 2026--2030, chương trình được định hướng chuyển trọng tâm ``từ số lượng sang chất lượng'', trong đó đẩy mạnh số hóa sản xuất, truy xuất nguồn gốc và tiêu thụ qua thương mại điện tử được xác định là một trong các trụ cột \parencite{vietnamnet_2025_ocop} --- đây là tín hiệu chính sách thuận lợi cho định hướng của Farmery, đồng thời cũng là nguồn cung sản phẩm có thương hiệu và câu chuyện xuất xứ, phù hợp để đưa lên một sàn TMĐT có tính năng truy xuất nguồn gốc.

Về chính sách, nông nghiệp được xác định là lĩnh vực ưu tiên chuyển đổi số từ Quyết định số 749/QĐ-TTg (2020) \parencite{qd749_2020} và Nghị quyết số 57-NQ/TW của Bộ Chính trị \parencite{nq57_tw}. Văn bản cập nhật nhất là \textbf{Quyết định số 5943/QĐ-BNNMT (31/12/2025)} của Bộ Nông nghiệp và Môi trường \parencite{botnmt_2025}, phê duyệt kế hoạch chuyển đổi số ngành với ba nội dung liên quan trực tiếp tới Farmery: xây dựng cơ sở dữ liệu dùng chung cho nông nghiệp, đưa dữ liệu về nguyên tắc chỉ khai báo một lần, và ứng dụng dữ liệu lớn cùng AI trong giám sát và dự báo sản xuất \parencite{qd5943_2025}. Đây là căn cứ cho thấy định hướng của đồ án phù hợp với chủ trương vĩ mô.

Về tiêu chuẩn chất lượng, VietGAP và GlobalGAP là hai chuẩn thực hành nông nghiệp tốt mà Farmery dùng làm căn cứ kiểm duyệt chứng nhận:
\begin{itemize}
    \item \textbf{VietGAP} (Vietnamese Good Agricultural Practices) --- tiêu chuẩn trong nước, khoảng 70 tiêu chí, phù hợp chuỗi cung ứng nội địa (siêu thị, bếp ăn tập thể, chợ đầu mối).
    \item \textbf{GlobalGAP} --- tiêu chuẩn quốc tế nghiêm ngặt hơn, 252 tiêu chuẩn (36 tiêu chí bắt buộc 100\%, 127 tiêu chí tối thiểu 95\%, 89 tiêu chí khuyến nghị), là điều kiện tiếp cận thị trường khó tính (EU, Nhật Bản, Hàn Quốc, Úc, Mỹ); sản phẩm đạt chuẩn được gán mã định danh GGN 13 chữ số trong cơ sở dữ liệu toàn cầu, truy xuất minh bạch hơn VietGAP.
\end{itemize}
Một cơ sở đạt GlobalGAP đã vượt chuẩn VietGAP nên không cần chứng nhận VietGAP riêng; VietGAP thường là bước đệm năng lực trước khi nông dân/HTX tiến tới GlobalGAP nếu có định hướng xuất khẩu \parencite{vietgap_globalgap}. Đây là căn cứ để Farmery thiết kế quy trình kiểm duyệt chứng nhận theo 2 cấp độ (nội địa/VietGAP và xuất khẩu/GlobalGAP) tương ứng với 2 nhóm khách mua lẻ và mua sỉ B2B.

\section{Cơ sở lý thuyết nền tảng}
\label{sec:lythuyet}

\subsection{Mô hình thương mại điện tử: hai trục phân loại}
\label{subsec:haitruc}

Thương mại điện tử được phân loại theo hai tiêu chí mà thực tế hay bị gộp làm một.

Tiêu chí thứ nhất là \textbf{bản chất các bên giao dịch}: B2B (Business to Business) giữa các doanh nghiệp, đặc trưng bởi đơn hàng số lượng lớn theo hợp đồng; B2C (Business to Consumer) giữa doanh nghiệp và người tiêu dùng cá nhân.

B2B2C (Business to Business to Consumer) là trường hợp chuỗi có ba mắt xích: một doanh nghiệp bán cho một doanh nghiệp khác, rồi doanh nghiệp thứ hai bán tiếp cho người tiêu dùng cuối.

Tiêu chí thứ hai --- và là tiêu chí quyết định với Farmery --- là \textbf{ai sở hữu hàng hóa trong lúc giao dịch diễn ra}. Mô hình \emph{third-party} (3P) để người bán giữ quyền sở hữu hàng cho tới khi giao; nền tảng chỉ trung gian và thu hoa hồng, không chịu rủi ro tồn kho. Mô hình \emph{first-party} (1P) thì ngược lại: chính nền tảng mua hàng vào, tự định giá và bán lại, hưởng chênh lệch nhưng gánh toàn bộ rủi ro hàng không bán hết.

Hai tiêu chí này \textbf{độc lập với nhau}, nên kết hợp được thành bốn trường hợp. Cùng là B2B2C nhưng Shopee theo hướng 3P (thương hiệu tự bán cho người tiêu dùng qua sàn), còn mảng bán lẻ của Amazon theo hướng 1P (Amazon mua từ thương hiệu rồi bán lại). Việc phân biệt hai trục này quan trọng vì chúng quyết định hai thứ khác nhau: trục thứ nhất quyết định hệ thống phải phục vụ những nhóm người dùng nào, trục thứ hai quyết định ai chịu rủi ro tài chính và ai chịu trách nhiệm chất lượng trước người mua.

Farmery vận dụng cả hai trục và khác nhau giữa hai kênh:
\begin{itemize}
    \item \textbf{Kênh bán sỉ --- B2B kết hợp 3P:} nhà cung cấp bán thẳng cho doanh nghiệp thu mua, Farmery đứng giữa.
    \item \textbf{Kênh bán lẻ --- B2B2C kết hợp 1P:} nhà cung cấp bán cho Farmery, Farmery bán tiếp cho người tiêu dùng, và sở hữu hàng trong suốt quá trình đó.
\end{itemize}
Lý do lựa chọn cùng cách hai kênh bổ trợ nhau được phân tích tại mục~\ref{sec:businessmodelcanvas}.

\subsection{Lý thuyết chuỗi cung ứng nông sản ngắn (SFSC) và disintermediation}

Chuỗi cung ứng thực phẩm ngắn (SFSC), theo định nghĩa của EU, là một chuỗi cung ứng liên quan đến một số lượng hạn chế các nhà điều hành kinh tế, cam kết hợp tác, phát triển kinh tế địa phương và duy trì mối quan hệ gần gũi về địa lý và xã hội giữa các nhà sản xuất thực phẩm, nhà chế biến và người tiêu dùng \parencite{tapchicongsan2018}. Có một số hình thức khác nhau của SFSC: bán trực tiếp từ nông dân đến người tiêu dùng cuối cùng (tại nông trại, chợ nông dân, giao hàng qua internet), các chương trình giao hộp sản phẩm, ``tự hái'', và nông nghiệp được cộng đồng hỗ trợ (CSA). Lợi ích chính của SFSC là tăng tính minh bạch và niềm tin cho người tiêu dùng, đồng thời mang lại lợi nhuận kinh tế tốt hơn và quyền tự chủ lớn hơn cho nhà sản xuất; hạn chế chính là khối lượng sản xuất/phân phối bị giới hạn, chi phí vận hành trên mỗi đơn vị sản phẩm cao hơn và khó mở rộng quy mô so với chuỗi cung ứng dài truyền thống.

Một khái niệm liên quan đến SFSC là disintermediation (loại bỏ trung gian) --- quá trình mà công nghệ số cho phép nhà sản xuất tiếp cận trực tiếp người mua cuối, cắt giảm các tầng trung gian truyền thống. Một ví dụ điển hình là chiến lược mà nhà bán lẻ máy tính trực tuyến Dell áp dụng vào đầu thế kỷ 21 \parencite{britannica_disintermediation}: công ty bán hàng hóa qua trang web của mình nhưng không có mặt trực tiếp tại các trung tâm mua sắm, giúp cung cấp hàng hóa với giá thấp hơn nhà bán lẻ truyền thống. Farmery về bản chất là một nỗ lực disintermediation có kiểm soát: không loại bỏ hoàn toàn trung gian (vẫn hợp tác với đơn vị vận chuyển), nhưng loại bỏ các tầng trung gian thương mại thuần túy hưởng chênh lệch giá mà không tạo thêm giá trị tương xứng như thương lái ép giá, đồng thời giữ được lợi ích quy mô của chuỗi cung ứng dài (tiếp cận thị trường rộng hơn) nhờ nền tảng số.

\subsection{Truy xuất nguồn gốc thực phẩm: chuẩn GS1 và vai trò của blockchain}
\label{subsec:gs1blockchain}

GS1 là tổ chức quốc tế phi lợi nhuận xây dựng các chuẩn định danh và trao đổi dữ liệu dùng chung toàn cầu cho chuỗi cung ứng, trong đó có bộ chuẩn dành riêng cho truy xuất nguồn gốc (GS1 Traceability Standard). GS1 định nghĩa cả mã định danh và phương tiện lưu trữ dữ liệu: GTIN cho sản phẩm ở mọi cấp độ đóng gói, GLN cho các bên và địa điểm, SSCC cho các đơn vị hậu cần và các khóa chuyên dụng cho tài sản và mối quan hệ. Sau khi có định danh, hệ thống ghi lại thông tin chuỗi cung ứng gắn với các định danh này, cho phép truy xuất thông tin lô hàng khi cần thu hồi hoặc xác minh xuất xứ \parencite{gs1_keys}. Farmery ứng dụng điều này và tạo ra các mã QR gắn với từng lô hàng, đại diện cho khả năng định danh nguồn gốc, ghi lại dữ liệu ứng với lô hàng đó.

Blockchain là công nghệ chuỗi khối phân tán, cho phép ghi chép và lưu trữ thông tin theo cách không thể thay đổi hoặc làm giả, nhờ đó đảm bảo tính minh bạch, tin cậy và bảo mật cao cho các giao dịch. Trong lĩnh vực thương mại điện tử, blockchain có thể hỗ trợ các bên (doanh nghiệp, nông dân, đơn vị vận chuyển) trong truy xuất nguồn gốc sản phẩm, tối ưu hóa quản lý chuỗi cung ứng và nâng cao hiệu quả giao dịch thương mại quốc tế.

Tuy nhiên, blockchain vấp phải một giới hạn căn bản khi áp dụng cho truy xuất nguồn gốc thực phẩm: nó chỉ bảo đảm dữ liệu không bị sửa \emph{sau khi đã ghi}, chứ không xác minh được dữ liệu đưa vào ban đầu có đúng hay không. Nông dân và cơ sở chế biến nhỏ vẫn nhập thông tin thủ công, nên gian lận tại nguồn hoặc tráo hàng trước khi đóng gói vẫn xảy ra mà blockchain không phát hiện được \parencite{bath_blockchain}. Vì vậy Farmery lưu nhật ký canh tác trên cơ sở dữ liệu tập trung theo nguyên lý định danh của GS1, kèm cơ chế bất biến sau khi công khai (NFR9.1).

Điều kiện để blockchain thực sự phát huy tác dụng thể hiện rõ ở trường hợp thành công được trích dẫn nhiều nhất: Walmart phối hợp với IBM Food Trust rút thời gian truy vết nguồn gốc một lô xoài bán tại Hoa Kỳ từ bảy ngày xuống còn 2,2 giây \parencite{walmart_ibm_foodtrust}. Kết quả này có một tiền đề mà Farmery không có: Walmart đủ lớn để buộc toàn bộ nhà cung cấp trong chuỗi nhập liệu đúng chuẩn lên cùng một hệ thống. Một sàn thương mại điện tử mới, với nhiều nhà cung cấp nhỏ lẻ phân tán, không có đòn bẩy đàm phán tương đương --- mà chính chất lượng dữ liệu đầu vào mới quyết định giá trị của bản ghi bất biến.

Lý do thường được viện dẫn để trì hoãn blockchain --- chi phí hạ tầng --- lại không đứng vững khi kiểm chứng. Một nghiên cứu so sánh chi phí hạ tầng công nghệ thông tin giữa mô hình tập trung và mô hình blockchain trên mười ba kịch bản triển khai cho chuỗi cung ứng gia cầm tại Thái Lan kết luận blockchain có chi phí hạ tầng thấp hơn khoảng 43\% so với cách làm tập trung đang vận hành \parencite{thai_broiler_blockchain_2025}. Báo cáo này vì vậy không lấy chi phí làm căn cứ để chưa dùng blockchain.

Căn cứ thật nằm ở nhóm rào cản khác. Một nghiên cứu năm 2025 về rào cản áp dụng blockchain cho chuỗi cung ứng thực phẩm dễ hư hỏng --- cùng đặc thù dễ hư hỏng của nông sản mà Farmery phục vụ --- xếp hạng mười bốn rào cản, ba vị trí cao nhất lần lượt là lo ngại bảo mật dữ liệu đi kèm mức độ trưởng thành công nghệ chưa đủ, nguy cơ xâm phạm quyền riêng tư dữ liệu, và thiếu hạ tầng số \parencite{perishable_blockchain_barriers_2025}. Rào cản thứ ba trùng khít với đặc điểm đã phân tích của nhóm nhà cung cấp tại mục~\ref{sec:doituongnguoidung}: hạn chế về kỹ năng số và thiết bị. Nói cách khác, đưa blockchain vào lúc này sẽ dựng thêm rào cản gia nhập cho đúng nhóm người dùng mà đề tài đặt mục tiêu hạ rào cản.

\subsection{Hệ thống đánh giá, uy tín và niềm tin (reputation, information asymmetry)}
\label{subsec:reputation}

Nền tảng lý thuyết cho vấn đề đảm bảo có một hệ thống đánh giá đủ uy tín được xây dựng dựa trên khái niệm bất cân xứng thông tin (information asymmetry), nổi bật với mô hình kinh điển ``The Market for Lemons'' của Akerlof (1970) \parencite{akerlof1970lemons}. Theo đó, khi người bán nắm rõ chất lượng thực tế của sản phẩm hơn người mua và người mua thiếu công cụ để kiểm chứng trước khi giao dịch, thị trường sẽ đối mặt với rủi ro ``lựa chọn đối nghịch'' (adverse selection) --- nơi hàng kém chất lượng dần đào thải hàng hóa tốt, vì người bán sản phẩm chất lượng cao không thể chứng minh giá trị thực để thuyết phục mức giá tương xứng. Đối với một sàn thương mại điện tử nông sản như Farmery, đây là bài toán cốt lõi cần giải quyết: nếu thiếu vắng các cơ chế minh bạch hóa, người tiêu dùng sẽ hoàn toàn mất phương hướng trong việc phân biệt nông sản ``sạch'' thực sự với các sản phẩm có mục đích marketing.

Hệ thống đánh giá và uy tín (reputation system) là cơ chế phổ biến để giảm bất cân xứng thông tin trên các sàn giao dịch trực tuyến, bằng cách tích lũy phản hồi từ các giao dịch trước đó (điểm số, bình luận, tỷ lệ hoàn thành đơn) thành một chỉ số uy tín công khai cho từng người bán \parencite{springer_article}. Các cơ chế này giúp thị trường vận hành hiệu quả hơn bằng cách tạo động lực cho người bán duy trì chất lượng để bảo vệ uy tín dài hạn, dù vẫn tồn tại vấn đề như tâm lý ngại đánh giá tiêu cực do sợ bị trả đũa, hoặc đánh giá tập trung ở nhóm khách hàng cực đoan. Với Farmery, hệ thống đánh giá cần kết hợp với dữ liệu truy xuất nguồn gốc khách quan và chứng nhận đã được xác minh như hai lớp bổ sung cho đánh giá chủ quan của người dùng, giúp giảm thiểu rủi ro đánh giá giả hoặc thao túng, đặc biệt quan trọng với nhóm khách B2B khi giá trị mỗi giao dịch lớn hơn nhiều so với B2C.

\subsection{Nền tảng công nghệ: kiến trúc web, RBAC, cổng thanh toán trung gian và escrow}

Về kiến trúc web, với một sàn thương mại điện tử đa vai trò như Farmery, để có thể phục vụ nhiều nhóm người dùng gồm nông dân/HTX, người tiêu dùng, doanh nghiệp B2B và quản trị viên, hệ thống có thể được xây dựng theo mô hình kiến trúc phân lớp hoặc kiến trúc hướng dịch vụ, tùy thuộc vào quy mô và yêu cầu mở rộng. Các thành phần chính bao gồm: lớp giao diện cung cấp web/app tương ứng với từng nhóm người dùng; lớp API và nghiệp vụ chịu trách nhiệm xử lý các quy trình kinh doanh; lớp dữ liệu sử dụng cơ sở dữ liệu quan hệ để lưu trữ thông tin người dùng, sản phẩm, đơn hàng, giao dịch và hợp đồng, kết hợp với hệ thống object storage cho hình ảnh, tài liệu và các chứng nhận liên quan. Ngoài ra, hệ thống có thể tích hợp với các dịch vụ bên thứ ba như cổng thanh toán, đơn vị vận chuyển và các hệ thống xác thực hoặc kiểm tra chứng nhận thông qua API khi cần thiết.

Đối với kiểm soát truy cập, Farmery áp dụng mô hình Role-Based Access Control (RBAC), trong đó quyền truy cập không được cấp trực tiếp cho từng người dùng mà được tổ chức thông qua các vai trò (role); người dùng được gán một hoặc nhiều vai trò, mỗi vai trò liên kết với một tập hợp các quyền tương ứng với chức năng nghiệp vụ. Mô hình RBAC được Sandhu và cộng sự hệ thống hóa thành một khung lý thuyết vào năm 1996 \parencite{sandhu1996rbac}. Với đặc thù của Farmery, RBAC phù hợp để quản lý tối thiểu bốn nhóm vai trò chính: nông dân/HTX, người tiêu dùng, doanh nghiệp thu mua và quản trị viên. Cách tổ chức này giúp áp dụng nguyên tắc đặc quyền tối thiểu (Principle of Least Privilege), qua đó hạn chế nguy cơ truy cập trái phép, sai sót trong vận hành và các rủi ro bảo mật.

Về thanh toán, trung gian thanh toán (payment intermediary) là các tổ chức được cấp phép cung cấp dịch vụ hỗ trợ thu hộ/chi hộ và chuyển tiền giữa các bên trong giao dịch điện tử mà không phải là ngân hàng trực tiếp của cả hai bên (ví như các cổng ví điện tử VNPay, Momo tại Việt Nam). Bên cạnh hình thức thanh toán thông thường, Farmery có thể xem xét áp dụng cơ chế escrow (ký quỹ trung gian) cho các giao dịch có giá trị lớn, đặc biệt là giao dịch B2B: khoản tiền của người mua được một bên trung gian giữ lại và chỉ được giải ngân cho người bán khi các điều kiện đã thỏa thuận được đáp ứng. Cơ chế này giúp giảm rủi ro cho cả hai phía: doanh nghiệp thu mua hạn chế nguy cơ thanh toán nhưng không nhận được hàng hóa đúng cam kết, trong khi nông dân/HTX có thêm sự đảm bảo rằng khoản tiền đã được người mua nộp và được giữ bởi bên trung gian.

\subsection{Mô hình doanh thu marketplace (business model)}
\label{subsec:businessmodel}

Theo Osterwalder và Pigneur, mô hình kinh doanh của một tổ chức được mô tả một cách có hệ thống qua 9 khối xây dựng (Business Model Canvas): phân khúc khách hàng, giá trị đề xuất, kênh phân phối, quan hệ khách hàng, dòng doanh thu, nguồn lực chính, hoạt động chính, đối tác chính và cơ cấu chi phí \parencite{osterwalder2010bmc}. Áp dụng khung này vào các sàn marketplace, dòng doanh thu (revenue streams) phổ biến gồm:
\begin{itemize}
    \item Phí hoa hồng trên mỗi giao dịch thành công (phổ biến nhất, tỷ lệ \% theo giá trị đơn hàng).
    \item Phí niêm yết/gói nổi bật để sản phẩm hiển thị ưu tiên.
    \item Phí dịch vụ giá trị gia tăng (xác thực chứng nhận nhanh, hỗ trợ chụp ảnh/marketing cho nhà cung cấp).
    \item Quảng cáo trong sàn.
\end{itemize}
Đa số marketplace giai đoạn đầu ưu tiên phí hoa hồng làm dòng doanh thu chính vì gắn trực tiếp với giá trị giao dịch thực tế phát sinh trên nền tảng, ít tạo rào cản gia nhập cho người bán mới so với phí niêm yết cố định. Mô hình doanh thu cụ thể của Farmery được phân tích chi tiết tại mục~\ref{sec:businessmodelcanvas}.

\subsection{Quản lý tồn kho hàng dễ hư hỏng và phối hợp với đơn vị vận chuyển thứ ba (3PL)}
\label{subsec:khobai}

Đối với hàng nông sản dễ hư hỏng, nguyên tắc quản lý tồn kho phổ biến là FEFO (First-Expired-First-Out) --- ưu tiên xuất bán lô hàng có hạn sử dụng/thời gian thu hoạch gần nhất trước, khác với nguyên tắc FIFO (First-In-First-Out) thường dùng cho hàng hóa không có hạn sử dụng rõ ràng. Việc quản lý tồn kho theo lô (batch) kết hợp với cảnh báo sớm khi lô hàng sắp hết hạn hoặc tồn kho xuống thấp giúp giảm tỷ lệ hao hụt --- vấn đề đã được phân tích tại Chương~\ref{chap:gioithieu} (Hình~\ref{fig:thuctrang}), nơi tỷ lệ tổn thất sau thu hoạch tại Việt Nam vẫn ở mức 20--40\% một phần do thiếu công cụ quản lý tồn kho hiệu quả ở khâu bán lẻ/bán sỉ.

Về vận chuyển, nhiều nền tảng thương mại điện tử không tự đầu tư đội xe hay kho bãi mà thuê ngoài đơn vị logistics bên thứ ba (3PL). Cách làm này giúp nền tảng tập trung nguồn lực vào công nghệ thay vì hạ tầng vật lý tốn kém --- đặc biệt là kho lạnh, vốn còn rất thiếu tại Việt Nam (Chương~\ref{chap:gioithieu}) --- nhưng đổi lại phải phụ thuộc vào chất lượng dịch vụ của đối tác. Farmery áp dụng mô hình này cho khâu vận chuyển ở cả hai kênh; ranh giới giữa phần nền tảng tự làm và phần thuê ngoài được nêu tại mục~\ref{sec:phamvi}.

\section{Ứng dụng AI trong phân tích dữ liệu và ra quyết định thương mại điện tử}

Bên cạnh các cơ sở lý thuyết về mô hình kinh doanh và niềm tin, Farmery còn tích hợp bốn nhóm bài toán trí tuệ nhân tạo (AI) nhằm hỗ trợ ra quyết định cho cả người dùng và đội ngũ vận hành, được thiết kế chi tiết tại mục~\ref{sec:thietkeAI} và hiện thực tại mục~\ref{sec:hienthucAI}.

\subsection{Hệ thống gợi ý sản phẩm (recommendation system)}
Hệ thống gợi ý là một trong những ứng dụng AI phổ biến nhất trong thương mại điện tử, giúp cá nhân hóa trải nghiệm mua sắm bằng cách đề xuất sản phẩm phù hợp dựa trên hành vi, lịch sử mua hàng hoặc đặc điểm của người dùng và sản phẩm tương tự. Theo khảo sát của Su và Khoshgoftaar (2009), hai hướng tiếp cận phổ biến là collaborative filtering (collaborative filtering) --- dựa trên sự tương đồng hành vi giữa các người dùng --- và content-based filtering (content-based filtering) --- dựa trên đặc điểm của sản phẩm đã tương tác trước đó; các hệ thống hiện đại thường kết hợp cả hai (hybrid) để giảm bớt vấn đề ``cold start'' (thiếu dữ liệu tương tác cho người dùng/sản phẩm mới) \parencite{su2009collaborative}. Với Farmery, hệ thống gợi ý có thể tận dụng đặc thù dữ liệu nông sản (mùa vụ, vùng trồng, chứng nhận) để gợi ý sản phẩm phù hợp thời điểm, giúp tăng khả năng tiêu thụ hàng tươi trước khi hết hạn sử dụng.

\subsection{Dự báo nhu cầu và giá nông sản}
Dự báo nhu cầu/giá là bài toán chuỗi thời gian (time-series forecasting): theo Hyndman và Athanasopoulos, phương pháp dự báo dựa trên dữ liệu lịch sử (sản lượng, giá, mùa vụ) để ước lượng xu hướng và tính mùa vụ trong tương lai gần, kết hợp giữa mô hình thống kê cổ điển (ARIMA, SARIMA) (ví dụ ARIMA, exponential smoothing) với các mô hình học máy khi dữ liệu đủ lớn \parencite{hyndman2021forecasting}. Với Farmery, dự báo giúp nông dân/HTX chủ động kế hoạch sản xuất và giúp hệ thống cảnh báo sớm nguy cơ dư cung/khan hiếm. Đây là hướng đã được kiểm chứng ở các nền tảng đối chứng: Ninjacart dùng học máy để dự báo nhu cầu và tối ưu vận chuyển \parencite{ninjacart}, còn Farm2Market công bố mô hình dự báo giá đạt độ chính xác khoảng 82\% \parencite{farm2market_2025}.

\subsection{Giám sát dấu hiệu bất thường}
Một sàn giao dịch luôn cần cơ chế phát hiện các dấu hiệu bất thường: đánh giá giả nhằm thao túng uy tín, mức giá lệch xa mặt bằng chung, hoặc hành vi đặt và hủy đơn liên tục. Theo khảo sát của Chandola, Banerjee và Kumar (2009), các kỹ thuật học máy phổ biến cho bài toán này gồm phương pháp thống kê, phân loại và phân cụm để phát hiện điểm dữ liệu lệch khỏi phân bố đã học được \parencite{chandola2009anomaly}.

Tuy nhiên, các kỹ thuật này đòi hỏi hai điều kiện mà một nền tảng mới chưa có: lưu lượng giao dịch thật đủ lớn để học được thế nào là ``bình thường'', và tập nhãn các trường hợp gian lận đã xác nhận để đo được chất lượng mô hình. Thiếu cả hai, mô hình học máy sẽ không kiểm chứng được bằng số đo --- mâu thuẫn với mục tiêu đánh giá đặt ra tại mục~\ref{sec:muctieudanhgia}. Vì vậy trong phạm vi đồ án, Farmery dùng \emph{luật ngưỡng} do quản trị viên cấu hình (ví dụ: số đánh giá năm sao từ cùng một người mua trong khoảng thời gian ngắn, tỷ lệ hủy đơn vượt mức, giá lệch quá một biên độ so với trung vị danh mục) làm lớp cảnh báo sơ bộ trước khi quản trị viên xem xét thủ công. Việc thay thế lớp luật ngưỡng này bằng mô hình học máy được đặt ở hướng phát triển (Chương~\ref{chap:ketluan}), khi nền tảng đã tích lũy đủ dữ liệu vận hành và nhãn từ chính phản hồi của quản trị viên.

\section{Kết chương}

Chương này đã hệ thống hóa các kiến thức nền tảng cần thiết cho đồ án:
\begin{itemize}
    \item \textbf{Đặc điểm mặt hàng:} phạm vi khái niệm nông sản, các đặc điểm phân biệt nông sản với hàng hóa thông thường và hệ quả của chúng lên thiết kế, cùng hệ thống chứng nhận để nông sản ra thị trường.
    \item \textbf{Bối cảnh và tiêu chuẩn:} định hướng chuyển đổi số nông nghiệp, chương trình OCOP và hai chuẩn VietGAP/GlobalGAP làm căn cứ kiểm duyệt chứng nhận.
    \item \textbf{Mô hình kinh doanh và chuỗi cung ứng:} các mô hình thương mại điện tử, chuỗi cung ứng thực phẩm ngắn và rút ngắn tầng trung gian, mô hình doanh thu, nguyên tắc xuất trước hàng hết hạn trước và hợp tác với đơn vị vận chuyển thứ ba --- làm nền cho ngữ cảnh nghiệp vụ tại mục~\ref{sec:ngucanh}.
    \item \textbf{Niềm tin và minh bạch:} truy xuất nguồn gốc theo GS1, lý do chưa dùng blockchain, lý thuyết bất cân xứng thông tin và hệ thống uy tín --- làm nền cho vấn đề V4 tại mục~\ref{sec:vande}.
    \item \textbf{Công nghệ nền tảng:} kiến trúc web phân lớp, RBAC, cổng thanh toán trung gian và cơ chế tạm giữ tiền.
    \item \textbf{Trí tuệ nhân tạo:} hai bài toán gợi ý sản phẩm và dự báo nhu cầu, giá; cùng lý do giám sát bất thường được hiện thực bằng luật ngưỡng thay vì học máy trong phạm vi đồ án.
\end{itemize}
Các kiến thức trên đủ bao quát phần phân tích yêu cầu và thiết kế ở Chương~\ref{chap:hethong} và Chương~\ref{chap:phantich}. Riêng các công nghệ hiện thực cụ thể (framework front-end/back-end, thư viện và mô hình AI) sẽ được so sánh và lựa chọn tại mục~\ref{sec:giaiphapcongnghe} sau khi nhóm chốt giải pháp.
